     4→async function getPrinter(): Promise<any> {
     5→  if (printerInstance) return printerInstance;
     6→  
     7→  // Dynamic import to handle ESM/CJS compatibility
     8→  const pdfmake = await import('pdfmake');
     9→  const PdfPrinter = pdfmake.default || pdfmake;
    10→  
    11→  const vfsFontsModule = await import('pdfmake/build/vfs_fonts');
    12→  const vfs = (vfsFontsModule as any).default?.pdfMake?.vfs || (vfsFontsModule as any).pdfMake?.vfs || (vfsFontsModule as any).default || vfsFontsModule;
    13→  
    14→  const fonts = {
    15→    Roboto: {
    16→      normal: 'Roboto-Regular.ttf',
    17→      bold: 'Roboto-Medium.ttf',
    18→      italics: 'Roboto-Italic.ttf',
    19→      bolditalics: 'Roboto-MediumItalic.ttf',
    20→    },
    21→  };
    22→  
    23→  const printer = new PdfPrinter(fonts);
    24→  printer.vfs = vfs;
    25→  printerInstance = printer;
    26→  return printerInstance;
    27→}

    84→async function pdfDocToBuffer(docDefinition: any): Promise<Buffer> {
    85→  const printer = await getPrinter();
    86→  return new Promise((resolve, reject) => {
    87→    const doc = printer.createPdfKitDocument(docDefinition);
    88→    const chunks: Buffer[] = [];
    89→    doc.on('data', (chunk: Buffer) => chunks.push(chunk));
    90→    doc.on('end', () => resolve(Buffer.concat(chunks)));
    91→    doc.on('error', (err: Error) => reject(err));
    92→    doc.end();
    93→  });
    94→}
