   118→function getFileType(fileUrl: string): "pdf" | "image" | "other" {
   119→  const lower = fileUrl.toLowerCase();
   120→  if (lower.includes(".pdf") || lower.startsWith("data:application/pdf")) return "pdf";
   121→  if (
   122→    lower.includes(".jpg") ||
   123→    lower.includes(".jpeg") ||
   124→    lower.includes(".png") ||
   125→    lower.startsWith("data:image/jpeg") ||
   126→    lower.startsWith("data:image/png")
   127→  ) return "image";
   128→  return "other";
   129→}

   131→// ─── PDF Canvas Preview Component ──────────────────────────────────
   132→// Renders PDF pages on a <canvas> using pdfjs-dist, avoiding iframe CORS issues.
   133→function PdfCanvasPreview({ url }: { url: string }) {
   134→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   135→  const [totalPages, setTotalPages] = useState(0);
   136→  const [currentPage, setCurrentPage] = useState(1);
   137→  const [renderError, setRenderError] = useState(false);
   138→
   139→  useEffect(() => {
   140→    if (!url) return;
   141→    let cancelled = false;
   142→    setRenderError(false);
   143→
   144→    const renderPdf = async (attempt = 0) => {
   145→      try {
   146→        const pdfjsLib = await import("pdfjs-dist");
   147→        pdfjsLib.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   148→
   149→        // Fetch PDF as ArrayBuffer for reliable rendering
   150→        const pdfResponse = await fetch(url);
   151→        const pdfArrayBuffer = await pdfResponse.arrayBuffer();
   152→        const loadingTask = pdfjsLib.getDocument({ data: pdfArrayBuffer });
   153→        const pdf = await loadingTask.promise;
   154→        if (cancelled) return;
   155→        setTotalPages(pdf.numPages);
   156→
   157→        const pageNum = Math.min(currentPage, pdf.numPages);
   158→        if (pageNum < 1) return;
   159→        const page = await pdf.getPage(pageNum);
   160→        if (cancelled) return;
   161→
   162→        const baseViewport = page.getViewport({ scale: 1 });
   163→        const maxWidth = 700;
   164→        const scale = Math.min(maxWidth / baseViewport.width, 2);
   165→        const viewport = page.getViewport({ scale });
   166→
   167→        const canvas = canvasRef.current;
   168→        if (!canvas) {
   169→          if (attempt < 10) {
   170→            setTimeout(() => { if (!cancelled) renderPdf(attempt + 1); }, 200);
   171→          }
   172→          return;
   173→        }
   174→
   175→        const context = canvas.getContext("2d");
   176→        if (!context) return;
   177→
   178→        canvas.height = viewport.height;
   179→        canvas.width = viewport.width;
   180→        await page.render({ canvasContext: context, viewport }).promise;
   181→      } catch (err) {
   182→        console.error("PDF render error:", err);
   183→        if (!cancelled) setRenderError(true);
   184→      }
   185→    };
   186→
   187→    renderPdf();
   188→    return () => { cancelled = true; };
   189→  }, [url, currentPage]);
   190→
   191→  if (renderError) {
   192→    return (
   193→      <div className="flex flex-col items-center justify-center h-[300px] gap-3">
   194→        <FileText className="size-12 text-muted-foreground" />
   195→        <p className="text-sm text-muted-foreground">Failed to render PDF preview.</p>
   196→      </div>
   197→    );
   198→  }
   199→
   200→  return (
   201→    <div className="space-y-2">
   202→      <div className="flex items-center justify-center gap-2">
   203→        <Button variant="ghost" size="sm" onClick={() => setCurrentPage((p) => Math.max(1, p - 1))} disabled={currentPage <= 1}>
   204→          ← Prev
   205→        </Button>
   206→        <span className="text-xs text-muted-foreground">Page {currentPage} of {totalPages}</span>
   207→        <Button variant="ghost" size="sm" onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))} disabled={currentPage >= totalPages}>
   208→          Next →
   209→        </Button>
   210→      </div>
   211→      <div className="flex justify-center overflow-auto max-h-[55vh]">
   212→        <canvas ref={canvasRef} className="block max-w-full" />
   213→      </div>
   214→    </div>
   215→  );
   216→}

   609→                ? "Upload your licenses, certifications, and other healthcare credentials to your secure vault."
   610→                : "Try changing the filter to see more credentials."}
   611→            </p>
   612→            {credentials.length === 0 && (
   613→              <Button
   614→                className="mt-4 gap-2"
   615→                onClick={() => setIsDialogOpen(true)}
   616→              >
   617→                <Plus className="size-4" />
   618→                Upload Your First Credential
   619→              </Button>
   620→            )}
   621→          </CardContent>
   622→        </Card>
   623→      ) : (
   624→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   625→          {filteredCredentials.map((credential) => (
   626→            <Card key={credential.id} className="group hover:shadow-md transition-shadow">
   627→              <CardContent className="p-4">
   628→                <div className="flex items-start gap-3">
   629→                  <div className="size-10 rounded-lg bg-primary/10 flex items-center justify-center shrink-0">
   630→                    <FileText className="size-5 text-primary" />
   631→                  </div>
   632→                  <div className="flex-1 min-w-0">
   633→                    <div className="flex items-start justify-between gap-2">
   634→                      <p className="font-medium text-sm truncate">
   635→                        {credential.document_name}
   636→                      </p>
   637→                      {/* Action buttons */}
   638→                      <div className="flex items-center gap-0.5 shrink-0">
   639→                        <Tooltip>
   640→                          <TooltipTrigger asChild>
   641→                            <Button
   642→                              variant="ghost"
   643→                              size="icon"
   644→                              className="size-7"
   645→                              onClick={() => openPreviewDialog(credential)}
   646→                            >
   647→                              <Eye className="size-3.5" />
   648→                            </Button>
   649→                          </TooltipTrigger>
   650→                          <TooltipContent>Preview</TooltipContent>
   651→                        </Tooltip>
   652→
   653→                        <Tooltip>
   654→                          <TooltipTrigger asChild>
   655→                            <Button
   656→                              variant="ghost"
   657→                              size="icon"
   658→                              className="size-7"
   659→                              onClick={() => openEditDialog(credential)}
   660→                            >
   661→                              <Pencil className="size-3.5" />
   662→                            </Button>
   663→                          </TooltipTrigger>
   664→                          <TooltipContent>Edit</TooltipContent>
   665→                        </Tooltip>
   666→
   667→                        <Tooltip>
   668→                          <TooltipTrigger asChild>
   669→                            <Button
   670→                              variant="ghost"
   671→                              size="icon"
   672→                              className="size-7"
   673→                              onClick={() => handleDownload(credential)}
   674→                              disabled={downloadingId === credential.id}
   675→                            >
   676→                              {downloadingId === credential.id ? (
   677→                                <Loader2 className="size-3.5 animate-spin" />
   678→                              ) : (
   679→                                <Download className="size-3.5" />
   680→                              )}
   681→                            </Button>
   682→                          </TooltipTrigger>
   683→                          <TooltipContent>Download</TooltipContent>
   684→                        </Tooltip>
   685→
   686→                        <AlertDialog>
   687→                          <Tooltip>
   688→                            <TooltipTrigger asChild>
   689→                              <AlertDialogTrigger asChild>
   690→                                <Button
   691→                                  variant="ghost"
   692→                                  size="icon"
   693→                                  className="size-7 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-950/30"
   694→                                  disabled={deletingId === credential.id}
   695→                                >
   696→                                  {deletingId === credential.id ? (
   697→                                    <Loader2 className="size-3.5 animate-spin" />
   698→                                  ) : (
   699→                                    <Trash2 className="size-3.5" />
   700→                                  )}
   701→                                </Button>
   702→                              </AlertDialogTrigger>
   703→                            </TooltipTrigger>
   704→                            <TooltipContent>Delete</TooltipContent>
   705→                          </Tooltip>
   706→                          <AlertDialogContent>
   707→                            <AlertDialogHeader>
   708→                              <AlertDialogTitle>Delete Credential</AlertDialogTitle>
   709→                              <AlertDialogDescription>
   710→                                Are you sure? This cannot be undone. The credential &quot;{credential.document_name}&quot; will be permanently removed from your vault.
   711→                              </AlertDialogDescription>
   712→                            </AlertDialogHeader>
   713→                            <AlertDialogFooter>
   714→                              <AlertDialogCancel>Cancel</AlertDialogCancel>
   715→                              <AlertDialogAction
   716→                                className="bg-red-600 hover:bg-red-700 text-white"
   717→                                onClick={() => handleDelete(credential.id)}
   718→                              >
   719→                                Delete
   720→                              </AlertDialogAction>
   721→                            </AlertDialogFooter>
   722→                          </AlertDialogContent>
   723→                        </AlertDialog>
   724→                      </div>
   725→                    </div>
   726→                    <div className="flex flex-wrap gap-1.5 mt-2">
   727→                      {getStatusBadge(credential.status)}
   728→                      {getVerificationBadge(credential.verification_status)}
   729→                    </div>
   730→                    {credential.expiration_date && (
   731→                      <div className="flex items-center gap-1 mt-2 text-xs text-muted-foreground">
   732→                        <Calendar className="size-3" />
   733→                        Expires {new Date(credential.expiration_date).toLocaleDateString()}
   734→                      </div>
   735→                    )}
   736→                    <div className="flex items-center gap-1 mt-1 text-xs text-muted-foreground">
   737→                      <Clock className="size-3" />
   738→                      Uploaded {new Date(credential.uploaded_at).toLocaleDateString()}
   739→                    </div>
   740→                  </div>
   741→                </div>
   742→              </CardContent>
   743→            </Card>
   744→          ))}
   745→        </div>
   746→      )}
   747→
   748→      {/* Edit Credential Dialog */}
   749→      <Dialog open={isEditDialogOpen} onOpenChange={setIsEditDialogOpen}>
   750→        <DialogContent className="sm:max-w-md">
   751→          <form onSubmit={handleEditSave}>
   752→            <DialogHeader>
   753→              <DialogTitle>Edit Credential</DialogTitle>
   754→              <DialogDescription>
   755→                Update your credential details
   756→              </DialogDescription>
   757→            </DialogHeader>
   758→            <div className="space-y-4 py-4">
   759→              <div className="space-y-2">
   760→                <Label htmlFor="editDocName">Document Name</Label>
   761→                <Input
   762→                  id="editDocName"
   763→                  placeholder="e.g., RN License, BLS Certification"
   764→                  value={editDocumentName}
   765→                  onChange={(e) => setEditDocumentName(e.target.value)}
   766→                  required
   767→                />
   768→              </div>
   769→              <div className="space-y-2">
   770→                <Label htmlFor="editExpDate">Expiration Date (Optional)</Label>
   771→                <Input
   772→                  id="editExpDate"
   773→                  type="date"
   774→                  value={editExpirationDate}
   775→                  onChange={(e) => setEditExpirationDate(e.target.value)}
   776→                />
   777→              </div>
   778→              <div className="flex items-center justify-between">
   779→                <div>
   780→                  <Label htmlFor="editReminder">30-Day Reminder</Label>
   781→                  <p className="text-xs text-muted-foreground">
   782→                    Get alerted 30 days before expiration
   783→                  </p>
   784→                </div>
   785→                <Switch
   786→                  id="editReminder"
   787→                  checked={editReminderEnabled}
   788→                  onCheckedChange={setEditReminderEnabled}
   789→                />
   790→              </div>
   791→            </div>
   792→            <DialogFooter>
   793→              <Button
   794→                type="button"
   795→                variant="outline"
   796→                onClick={() => setIsEditDialogOpen(false)}
   797→                disabled={isSavingEdit}
   798→              >
   799→                Cancel
   800→              </Button>
   801→              <Button type="submit" disabled={isSavingEdit} className="gap-2">
   802→                {isSavingEdit ? (
   803→                  <>
   804→                    <Loader2 className="size-4 animate-spin" />
   805→                    Saving...
   806→                  </>
   807→                ) : (
   808→                  <>
   809→                    <CheckCircle2 className="size-4" />
   810→                    Save Changes
   811→                  </>
   812→                )}
   813→              </Button>
   814→            </DialogFooter>
   815→          </form>
   816→        </DialogContent>
   817→      </Dialog>
   818→
   819→      {/* Preview Credential Dialog */}
   820→      <Dialog open={isPreviewDialogOpen} onOpenChange={setIsPreviewDialogOpen}>
   821→        <DialogContent className="sm:max-w-3xl max-h-[80vh]">
   822→          <DialogHeader>
   823→            <DialogTitle className="flex items-center gap-2">
   824→              <Eye className="size-5" />
   825→              {previewCredential?.document_name || "Preview"}
   826→            </DialogTitle>
   827→            <DialogDescription>
   828→              File preview — close this dialog when done
   829→            </DialogDescription>
   830→          </DialogHeader>
   831→          <div className="min-h-[300px] max-h-[60vh] overflow-auto rounded-lg border bg-muted/30">
   832→            {isLoadingPreview ? (
   833→              <div className="flex items-center justify-center h-[300px]">
   834→                <Loader2 className="size-8 animate-spin text-muted-foreground" />
   835→              </div>
   836→            ) : previewUrl ? (
   837→              (() => {
   838→                const fileType = getFileType(previewCredential?.file_url || "");
   839→                if (fileType === "pdf") {
   840→                  return <PdfCanvasPreview url={previewUrl} />;
   841→                }
   842→                if (fileType === "image") {
   843→                  return (
   844→                    <div className="flex items-center justify-center p-4">
   845→                      <img
   846→                        src={previewUrl}
   847→                        alt={previewCredential?.document_name || "Credential preview"}
   848→                        className="max-w-full max-h-[60vh] object-contain rounded"
   849→                      />
   850→                    </div>
   851→                  );
   852→                }
   853→                return (
   854→                  <div className="flex flex-col items-center justify-center h-[300px] gap-3">
   855→                    <FileText className="size-12 text-muted-foreground" />
   856→                    <p className="text-sm text-muted-foreground text-center">
   857→                      Preview not available for this file type.
   858→                    </p>
   859→                    <Button
   860→                      variant="outline"
   861→                      size="sm"
   862→                      className="gap-2"
   863→                      onClick={() => {
   864→                        if (previewCredential) handleDownload(previewCredential);
   865→                      }}
   866→                    >
   867→                      <Download className="size-4" />
   868→                      Download File
   869→                    </Button>
   870→                  </div>
   871→                );
   872→              })()
   873→            ) : (
   874→              <div className="flex items-center justify-center h-[300px]">
   875→                <p className="text-sm text-muted-foreground">Failed to load preview</p>
   876→              </div>
   877→            )}
   878→          </div>
   879→        </DialogContent>
   880→      </Dialog>
   881→    </div>
   882→  );
   883→}
