    69→export default function SuperAdminTemplateEditorPage({ params }: { params: Promise<{ id: string }> }) {
    70→  const router = useRouter();
    71→  const [templateId, setTemplateId] = useState<string>("");
    72→  const [template, setTemplate] = useState<any>(null);
    73→  const [loading, setLoading] = useState(true);
    74→  const [saving, setSaving] = useState(false);
    75→  const [templateName, setTemplateName] = useState("");
    76→  const [templateDescription, setTemplateDescription] = useState("");
    77→  const [signFields, setSignFields] = useState<SignField[]>([]);
    78→  const [customVariables, setCustomVariables] = useState<Array<{ key: string; label: string }>>([]);
    79→  const [newVarKey, setNewVarKey] = useState("");
    80→  const [newVarLabel, setNewVarLabel] = useState("");
    81→  const [signerSlots, setSignerSlots] = useState<Array<{ index: number; label: string }>>([
    82→    { index: 0, label: "Signer 1" },
    83→    { index: 1, label: "Signer 2" },
    84→    { index: 2, label: "Signer 3" },
    85→  ]);
    86→  const [showVariablesPanel, setShowVariablesPanel] = useState(false);
    87→  const [showSignersPanel, setShowSignersPanel] = useState(false);
    88→  const [showHeaderFooter, setShowHeaderFooter] = useState(true);
    89→  const autoSaveTimer = useRef<NodeJS.Timeout | null>(null);
    90→
    91→  // Unwrap params
    92→  useEffect(() => {
    93→    params.then((p) => setTemplateId(p.id));
    94→  }, [params]);
    95→
    96→  // Initialize TipTap editor
    97→  const editor = useEditor({
    98→    extensions: [
    99→      StarterKit.configure({
   100→        heading: { levels: [1, 2, 3] },
   101→      }),
   102→      FontFamily,
   103→      TextStyle,
   104→      Color,
   105→      FontSize,
   106→      LineHeight,
   107→      ParagraphSpacing,
   108→      PageBreak,
   109→      SignFieldExtension,
   110→      Highlight.configure({ multicolor: true }),
   111→      TextAlign.configure({ types: ["heading", "paragraph"] }),
   112→      Underline,
   113→      Subscript,
   114→      Superscript,
   115→      Table.configure({ resizable: true }),
   116→      TableRow,
   117→      TableCell,
   118→      TableHeader,
   119→      Image,
   120→      TaskList,
   121→      TaskItem.configure({ nested: true }),
   122→      CharacterCount,
   123→      Placeholder.configure({ placeholder: "Start building your template..." }),
   124→    ],
   125→    content: "",
   126→    onUpdate: ({ editor }) => {
   127→      if (autoSaveTimer.current) clearTimeout(autoSaveTimer.current);
   128→      autoSaveTimer.current = setTimeout(() => {
   129→        handleAutoSave(editor.getJSON());
   130→      }, 5000);
   131→    },
   132→    editorProps: {
   133→      attributes: {
   134→        class: "prose prose-sm max-w-none focus:outline-none min-h-[500px] px-8 py-6",
   135→      },
   136→    },
   137→  });
   138→
   139→  // Fetch template data
   140→  const fetchTemplate = useCallback(async () => {
   141→    if (!templateId) return;
   142→    try {
   143→      setLoading(true);
   144→      const res = await fetch(`/api/superadmin/vaultsign/templates/${templateId}`);
   145→      if (!res.ok) throw new Error("Failed to fetch template");
   146→      const data = await res.json();
   147→      setTemplate(data);
   148→      setTemplateName(data.name || "");
   149→      setTemplateDescription(data.description || "");
   150→
   151→      // Parse sign fields
   152→      if (data.predefined_sign_fields) {
   153→        const fields = typeof data.predefined_sign_fields === "string"
   154→          ? JSON.parse(data.predefined_sign_fields)
   155→          : data.predefined_sign_fields;
   156→        setSignFields(Array.isArray(fields) ? fields : []);
   157→      }
   158→
   159→      // Parse placeholder variables
   160→      if (data.placeholder_variables) {
   161→        const vars = typeof data.placeholder_variables === "string"
   162→          ? JSON.parse(data.placeholder_variables)
   163→          : data.placeholder_variables;
   164→        setCustomVariables(vars.filter((v: any) => v.category === "custom"));
   165→      }
   166→
   167→      // Parse show_header_footer
   168→      if (data.show_header_footer !== undefined) {
   169→        setShowHeaderFooter(data.show_header_footer);
   170→      } else if (data.header_config || data.footer_config) {
   171→        // Backward compat
   172→        try {
   173→          const hc = typeof data.header_config === "string" ? JSON.parse(data.header_config) : data.header_config;
   174→          const fc = typeof data.footer_config === "string" ? JSON.parse(data.footer_config) : data.footer_config;
   175→          const anyHeaderOn = hc && (hc.show_logo || hc.show_company_name || hc.show_contact || hc.show_address || hc.show_document_title);
   176→          const anyFooterOn = fc && (fc.show_rights_reserved || fc.show_powered_by || fc.show_page_numbers);
   177→          setShowHeaderFooter(!!anyHeaderOn || !!anyFooterOn);
   178→        } catch {
   179→          setShowHeaderFooter(true);
   180→        }
   181→      }
   182→
   183→      // Set editor content
   184→      if (editor && data.tiptap_content) {
   185→        try {
   186→          const rawContent = typeof data.tiptap_content === "string"
   187→            ? data.tiptap_content
   188→            : JSON.stringify(data.tiptap_content);
   189→
   190→          try {
   191→            const parsed = JSON.parse(rawContent);
   192→            if (parsed.type === "doc" && parsed.content) {
   193→              editor.commands.setContent(parsed);
   194→            } else {
   195→              editor.commands.setContent(rawContent);
   196→            }
   197→          } catch {
   198→            editor.commands.setContent(rawContent);
   199→          }
   200→        } catch (setContentErr) {
   201→          console.error("Failed to set editor content:", setContentErr);
   202→          try {
   203→            editor.commands.setContent(data.tiptap_content);
   204→          } catch {
   205→            // Give up silently
   206→          }
   207→        }
   208→      }
   209→    } catch (err) {
   210→      console.error("Fetch error:", err);
   211→      toast.error("Failed to load template");
   212→    } finally {
   213→      setLoading(false);
   214→    }
   215→  }, [templateId, editor]);
   216→
   217→  useEffect(() => {
   218→    fetchTemplate();
   219→  }, [fetchTemplate]);
   220→
   221→  // Auto-save
   222→  const handleAutoSave = async (content?: any) => {
   223→    if (!templateId || !editor) return;
   224→    try {
   225→      setSaving(true);
   226→      const editorContent = content || editor.getJSON();
   227→      await fetch(`/api/superadmin/vaultsign/templates/${templateId}`, {
   228→        method: "PATCH",
   229→        headers: { "Content-Type": "application/json" },
   230→        body: JSON.stringify({
   231→          name: templateName,
   232→          description: templateDescription,
   233→          tiptap_content: JSON.stringify(editorContent),
   234→          predefined_sign_fields: signFields,
   235→          placeholder_variables: [
   236→            ...SYSTEM_VARIABLES,
   237→            ...customVariables.map((v) => ({ ...v, category: "custom" as const })),
   238→          ],
   239→          show_header_footer: showHeaderFooter,
   240→        }),
   241→      });
   242→    } catch (err) {
   243→      console.error("Auto-save error:", err);
   244→    } finally {
   245→      setSaving(false);
   246→    }
   247→  };
   248→
   249→  // Manual save
   250→  const handleSave = async () => {
   251→    if (autoSaveTimer.current) clearTimeout(autoSaveTimer.current);
   252→    try {
   253→      setSaving(true);
   254→      const editorContent = editor?.getJSON();
   255→      const res = await fetch(`/api/superadmin/vaultsign/templates/${templateId}`, {
   256→        method: "PATCH",
   257→        headers: { "Content-Type": "application/json" },
   258→        body: JSON.stringify({
   259→          name: templateName,
   260→          description: templateDescription,
   261→          tiptap_content: JSON.stringify(editorContent),
   262→          predefined_sign_fields: signFields,
   263→          placeholder_variables: [
   264→            ...SYSTEM_VARIABLES,
   265→            ...customVariables.map((v) => ({ ...v, category: "custom" as const })),
   266→          ],
   267→          show_header_footer: showHeaderFooter,
   268→        }),
   269→      });
   270→      if (res.ok) {
   271→        toast.success("Template saved");
   272→      } else {
   273→        const data = await res.json().catch(() => ({}));
   274→        toast.error(data.error || "Failed to save template");
   275→      }
   276→    } catch {
   277→      toast.error("Failed to save template");
   278→    } finally {
   279→      setSaving(false);
   280→    }
   281→  };
   282→
   283→  // Insert variable at cursor
   284→  const insertVariable = (varLabel: string) => {
   285→    if (!editor) return;
   286→    editor.chain().focus().insertContent(varLabel).run();
   287→  };
   288→
   289→  // Add sign field
   290→  const addSignField = (type: SignFieldType, signerIndex: number) => {
   291→    const newField: SignField = {
   292→      id: `field_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
   293→      type,
   294→      page: 1,
   295→      x_percent: 0,
   296→      y_percent: 0,
   297→      width_percent: 20,
   298→      height_percent: 3,
   299→      assigned_to_signer_index: signerIndex,
   300→      label: FIELD_TYPE_LABELS[type],
   301→      required: true,
   302→      value: null,
   303→    };
   304→    setSignFields([...signFields, newField]);
   305→
   306→    if (editor) {
   307→      editor.chain().focus().insertSignField({
   308→        fieldType: type,
   309→        assignedToSignerIndex: signerIndex,
   310→        signerLabel: signerSlots.find(s => s.index === signerIndex)?.label || `Signer ${signerIndex + 1}`,
   311→        fieldId: newField.id,
   312→      }).run();
   313→    }
   314→  };
   315→
   316→  // Remove sign field
   317→  const removeSignField = (fieldId: string) => {
   318→    setSignFields(signFields.filter((f) => f.id !== fieldId));
   319→  };
   320→
   321→  // Add custom variable
   322→  const addCustomVariable = () => {
   323→    if (!newVarKey || !newVarLabel) return;
   324→    setCustomVariables([...customVariables, { key: newVarKey, label: newVarLabel }]);
   325→    setNewVarKey("");
   326→    setNewVarLabel("");
   327→  };
   328→
   329→  // Add signer slot
   330→  const addSignerSlot = () => {
   331→    setSignerSlots([...signerSlots, { index: signerSlots.length, label: `Signer ${signerSlots.length + 1}` }]);
   332→  };
   333→
   334→  // Remove signer slot
   335→  const removeSignerSlot = (index: number) => {
   336→    setSignerSlots(signerSlots.filter((s) => s.index !== index));
   337→    setSignFields(signFields.filter((f) => f.assigned_to_signer_index !== index));
   338→  };
   339→
   340→  // Loading skeleton
   341→  if (loading) {
   342→    return (
   343→      <div className="min-h-screen bg-[#F8F7F4] flex flex-col">
   344→        <div className="bg-white border-b border-[#E5E7EB] px-4 py-3 flex items-center gap-3">
   345→          <Skeleton className="h-8 w-16" />
   346→          <Skeleton className="h-6 w-px" />
   347→          <Skeleton className="h-6 w-40" />
   348→          <Skeleton className="h-5 w-24" />
   349→          <div className="ml-auto flex items-center gap-2">
   350→            <Skeleton className="h-8 w-24" />
   351→            <Skeleton className="h-8 w-24" />
   352→          </div>
   353→        </div>
   354→        <div className="bg-white border-b border-[#E5E7EB] px-4 py-2 flex items-center gap-1">
   355→          {Array.from({ length: 14 }).map((_, i) => (
   356→            <Skeleton key={i} className="h-8 w-8 rounded" />
   357→          ))}
   358→          <Skeleton className="h-8 w-[120px] rounded ml-1" />
   359→        </div>
   360→        <div className="flex-1 flex overflow-hidden">
   361→          <div className="hidden lg:flex w-64 border-r border-[#E5E7EB] bg-white flex-col">
   362→            <div className="p-3 border-b border-[#E5E7EB]">
   363→              <Skeleton className="h-5 w-24" />
   364→              <Skeleton className="h-3 w-36 mt-1" />
   365→            </div>
   366→            <div className="p-3 space-y-2">
   367→              {Array.from({ length: 6 }).map((_, i) => (
   368→                <Skeleton key={i} className="h-7 w-full rounded" />
   369→              ))}
   370→            </div>
   371→          </div>
   372→          <div className="flex-1 overflow-y-auto bg-white p-6">
   373→            <div className="max-w-3xl mx-auto bg-white rounded-2xl border border-[#E5E7EB] shadow-sm min-h-[800px] p-8 space-y-4">
   374→              <Skeleton className="h-8 w-3/4" />
   375→              <Skeleton className="h-4 w-full" />
   376→              <Skeleton className="h-4 w-full" />
   377→              <Skeleton className="h-4 w-5/6" />
   378→            </div>
   379→          </div>
   380→          <div className="hidden lg:flex w-72 border-l border-[#E5E7EB] bg-white flex-col">
   381→            <div className="p-3 border-b border-[#E5E7EB]">
   382→              <Skeleton className="h-5 w-32" />
   383→            </div>
   384→            <div className="p-3 space-y-3">
   385→              {Array.from({ length: 3 }).map((_, i) => (
   386→                <Skeleton key={i} className="h-14 w-full rounded-lg" />
   387→              ))}
   388→            </div>
   389→          </div>
   390→        </div>
   391→      </div>
   392→    );
   393→  }
   394→
   395→  // Variables panel content
   396→  const variablesPanelContent = (
   397→    <>
   398→      <ScrollArea className="flex-1">
   399→        <div className="p-3">
   400→          <h4 className="text-xs font-semibold text-[#6B7280] uppercase mb-2">Template Variables</h4>
   401→          <p className="text-[10px] text-[#9CA3AF] mb-2">Click to insert a placeholder into the template</p>
   402→          <div className="space-y-1.5">
   403→            {TEMPLATE_VARIABLES.map((v) => (
   404→              <button
   405→                key={v.key}
   406→                onClick={() => { insertVariable(v.label); setShowVariablesPanel(false); }}
   407→                className="w-full text-left px-2.5 py-1.5 rounded-lg text-xs bg-[#F0FDF4] hover:bg-[#DCFCE7] text-[#166534] font-mono font-medium transition-colors border border-transparent hover:border-[#166534]/20"
   408→              >
   409→                {v.label}
   410→              </button>
   411→            ))}
   412→          </div>
   413→
   414→          <Separator className="my-3" />
   415→
   416→          <h4 className="text-xs font-semibold text-[#6B7280] uppercase mb-2">System Variables</h4>
   417→          <div className="space-y-1.5">
   418→            {SYSTEM_VARIABLES.map((v) => (
   419→              <button
   420→                key={v.key}
   421→                onClick={() => { insertVariable(`{{${v.key}}}`); setShowVariablesPanel(false); }}
   422→                className="w-full text-left px-2.5 py-1.5 rounded-lg text-xs bg-[#F0FDF4] hover:bg-[#DCFCE7] text-[#166534] font-mono font-medium transition-colors border border-transparent hover:border-[#166534]/20"
   423→              >
   424→                {`{{${v.key}}}`}
   425→              </button>
   426→            ))}
   427→          </div>
   428→
   429→          <Separator className="my-3" />
   430→
   431→          <h4 className="text-xs font-semibold text-[#6B7280] uppercase mb-2">Custom Variables</h4>
   432→          <div className="space-y-1.5">
   433→            {customVariables.map((v) => (
   434→              <div key={v.key} className="flex items-center gap-1">
   435→                <button
   436→                  onClick={() => { insertVariable(`{{${v.key}}}`); setShowVariablesPanel(false); }}
   437→                  className="flex-1 text-left px-2.5 py-1.5 rounded-lg text-xs bg-[#EFF6FF] hover:bg-[#DBEAFE] text-[#1D4ED8] font-mono font-medium transition-colors"
   438→                >
   439→                  {`{{${v.key}}}`}
   440→                </button>
   441→                <Button
   442→                  variant="ghost"
   443→                  size="sm"
   444→                  className="h-6 w-6 p-0 text-[#9CA3AF] hover:text-[#DC2626]"
   445→                  onClick={() => setCustomVariables(customVariables.filter((cv) => cv.key !== v.key))}
   446→                >
   447→                  <X className="h-3 w-3" />
   448→                </Button>
   449→              </div>
   450→            ))}
   451→          </div>
   452→
   453→          <div className="mt-3 p-2 rounded-lg border border-dashed border-[#E5E7EB]">
   454→            <Input
   455→              placeholder="Variable key"
   456→              value={newVarKey}
   457→              onChange={(e) => setNewVarKey(e.target.value)}
   458→              className="h-7 text-xs mb-1.5"
   459→            />
   460→            <Input
   461→              placeholder="Display label"
   462→              value={newVarLabel}
   463→              onChange={(e) => setNewVarLabel(e.target.value)}
   464→              className="h-7 text-xs mb-1.5"
   465→            />
   466→            <Button
   467→              size="sm"
   468→              className="w-full h-7 text-xs bg-[#166534] hover:bg-[#14532D]"
   469→              onClick={addCustomVariable}
   470→              disabled={!newVarKey || !newVarLabel}
   471→            >
   472→              <Plus className="h-3 w-3 mr-1" /> Add
   473→            </Button>
   474→          </div>
   475→        </div>
   476→      </ScrollArea>
   477→    </>
   478→  );
   479→
   480→  // Sign fields panel content
   481→  const signersPanelContent = (
   482→    <>
   483→      <ScrollArea className="flex-1">
   484→        <div className="p-3 space-y-4">
   485→          {/* Header & Footer Settings */}
   486→          <div>
   487→            <div className="flex items-center justify-between px-2 py-2 rounded-lg bg-[#F8F7F4] border border-[#E5E7EB]">
   488→              <div>
   489→                <span className="text-xs font-medium text-[#374151]">Header & Footer</span>

   491→              </div>
   492→              <button 
   493→                onClick={() => setShowHeaderFooter(!showHeaderFooter)} 
   494→                className={`w-9 h-5 rounded-full transition-colors ${showHeaderFooter ? 'bg-[#166534]' : 'bg-[#D1D5DB]'}`}
   495→              >
   496→                <div className={`w-4 h-4 rounded-full bg-white transition-transform shadow-sm ${showHeaderFooter ? 'translate-x-4' : 'translate-x-0.5'}`} />
   497→              </button>
   498→            </div>
   499→          </div>
   500→
   501→          <Separator />
   502→
   503→          {/* Signer Slots */}
