     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef } from "react";
     4→import { useRouter } from "next/navigation";
     5→import { useSession } from "next-auth/react";
     6→import { Card, CardContent } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Input } from "@/components/ui/input";
     9→import { Label } from "@/components/ui/label";
    10→import { Badge } from "@/components/ui/badge";
    11→import { Textarea } from "@/components/ui/textarea";
    12→import { Progress } from "@/components/ui/progress";
    13→import { Skeleton } from "@/components/ui/skeleton";
    14→import {
    15→  Lock,
    16→  Loader2,
    17→  CheckCircle2,
    18→  ShieldCheck,
    19→  ChevronRight,
    20→} from "@/lib/icons";
    21→import { toast } from "sonner";
    22→import Link from "next/link";
    23→
    24→/* ─── Types ─────────────────────────────────────────────────────────── */
    25→interface SkillItem {
    26→  id: number;
    27→  skillName: string;
    28→  category: string;
    29→  questionType: string;
    30→  sortOrder: number;
    31→  hasNaOption: boolean;
    32→}
    33→
    34→interface RatingItem {
    35→  id: number;
    36→  ratingValue: string | null;
    37→  isNa: boolean;
    38→}
    39→
    40→interface ChecklistData {
    41→  checklistRequest: {
    42→    id: number;
    43→    status: string;
    44→    completionPct: number;
    45→    createdAt: string;
    46→  };
    47→  template: {
    48→    id: number;
    49→    name: string;
    50→    profession: string;
    51→    specialty: string;
    52→  };
    53→  client: {
    54→    firstName: string | null;
    55→    lastName: string | null;
    56→    organizationName: string | null;
    57→  };
    58→  candidateResponse: {
    59→    id: number;
    60→    status: string;
    61→    submittedAt: string | null;
    62→    digitalSignature: string | null;
    63→    candidateNameSigned: string | null;
    64→    validUntil: string | null;
    65→  } | null;
    66→  skills: SkillItem[];
    67→  ratings: Record<number, RatingItem>;
    68→  ratedSkills: number;
    69→  totalSkills: number;
    70→  completionPct: number;
    71→}
    72→
    73→interface SignaturePadInstance {
    74→  clear(): void;
    75→  isEmpty(): boolean;
    76→  toDataURL(type?: string): string;
    77→  toData(): unknown[];
    78→  fromData(data: unknown[]): void;
    79→  off(): void;
    80→}
    81→
    82→/* ─── Rating color map ──────────────────────────────────────────────── */
    83→const RATING_COLORS: Record<number, { bg: string; text: string; border: string }> = {
    84→  1: { bg: "#FEE2E2", text: "#DC2626", border: "#DC2626" },
    85→  2: { bg: "#FEF9C3", text: "#CA8A04", border: "#CA8A04" },
    86→  3: { bg: "#DBEAFE", text: "#2563EB", border: "#2563EB" },
    87→  4: { bg: "#DCFCE7", text: "#166534", border: "#166534" },
    88→  5: { bg: "#166534", text: "#FFFFFF", border: "#166534" },
    89→};
    90→
    91→const RATING_LABELS: Record<number, string> = {
    92→  1: "No Experience",
    93→  2: "Beginner",
    94→  3: "Competent",
    95→  4: "Proficient",
    96→  5: "Expert",
    97→};
    98→
    99→/* ─── Helper: check if a rating is effectively rated ─────────────────── */
   100→function isRatingDone(rating: RatingItem | undefined): boolean {
   101→  if (!rating) return false;
   102→  if (rating.isNa) return true;
   103→  return rating.ratingValue !== null && rating.ratingValue !== "";
   104→}
   105→
   106→/* ─── Component ─────────────────────────────────────────────────────── */
   107→export default function ChecklistDetailPage({
   108→  params,
   109→}: {
   110→  params: Promise<{ id: string }>;
   111→}) {
   112→  const router = useRouter();
   113→  const { data: session, status: sessionStatus } = useSession();
   114→  const [id, setId] = useState<string>("");
   115→  const [data, setData] = useState<ChecklistData | null>(null);
   116→  const [isLoading, setIsLoading] = useState(true);
   117→  const [ratings, setRatings] = useState<Record<number, RatingItem>>({});
   118→  const [candidateNameSigned, setCandidateNameSigned] = useState("");
   119→  const [isSubmitting, setIsSubmitting] = useState(false);
   120→  const [autoSaveStatus, setAutoSaveStatus] = useState<"idle" | "saving" | "saved">("idle");
   121→
   122→  // Signature pad
   123→  const sigCanvasRef = useRef<HTMLCanvasElement>(null);
   124→  const sigPadRef = useRef<SignaturePadInstance | null>(null);
   125→  const [hasSigned, setHasSigned] = useState(false);
   126→  const sigPadInitialized = useRef(false);
   127→
   128→  // Unwrap params
   129→  useEffect(() => {
   130→    params.then((p) => setId(p.id));
   131→  }, [params]);
   132→
   133→  // Fetch checklist data
   134→  const fetchChecklist = useCallback(async () => {
   135→    if (!id) return;
   136→    try {
   137→      const res = await fetch(`/api/candidate/checklists/${id}`);
   138→      if (!res.ok) throw new Error("Failed to fetch checklist");
   139→      const checklistData = await res.json();
   140→      setData(checklistData);
   141→      setRatings(checklistData.ratings || {});
   142→
   143→      // Restore previously signed name if re-visiting
   144→      if (checklistData.candidateResponse?.candidateNameSigned) {
   145→        setCandidateNameSigned(checklistData.candidateResponse.candidateNameSigned);
   146→      }
   147→    } catch {
   148→      toast.error("Failed to load checklist. Please try again.");
   149→    } finally {
   150→      setIsLoading(false);
   151→    }
   152→  }, [id]);
   153→
   154→  useEffect(() => {
   155→    fetchChecklist();
   156→  }, [fetchChecklist]);
   157→
   158→  // Security: redirect if not authenticated
   159→  useEffect(() => {
   160→    if (sessionStatus === "unauthenticated") {
   161→      toast.error("Please sign in to access this checklist.");
   162→      router.push("/dashboard");
   163→    }
   164→  }, [sessionStatus, router]);
   165→
   166→  /* ─── Derived: allSkillsRated ────────────────────────────────────── */
   167→  const allSkillsRated =
   168→    (data?.skills.length ?? 0) > 0 &&
   169→    data!.skills.every((s) => isRatingDone(ratings[s.id]));
   170→
   171→  // Initialize signature pad – only once when canvas becomes available
   172→  useEffect(() => {
   173→    if (!allSkillsRated) return;
   174→    if (sigPadInitialized.current) return;
   175→
   176→    let destroyed = false;
   177→
   178→    import("signature_pad").then(({ default: SignaturePad }) => {
   179→      if (destroyed) return;
   180→      requestAnimationFrame(() => {
   181→        if (destroyed) return;
   182→        const canvas = sigCanvasRef.current;
   183→        if (!canvas) return;
   184→        if (data?.candidateResponse?.status === "submitted") return;
   185→
   186→        const ratio = Math.max(window.devicePixelRatio || 1, 1);
   187→        canvas.width = canvas.offsetWidth * ratio;
   188→        canvas.height = canvas.offsetHeight * ratio;
   189→        const ctx = canvas.getContext("2d");
   190→        if (ctx) ctx.scale(ratio, ratio);
   191→
   192→        sigPadRef.current = new SignaturePad(canvas, {
   193→          backgroundColor: "rgb(255,255,255)",
   194→          penColor: "rgb(22,101,52)",
   195→          onEnd: () => {
   196→            const pad = sigPadRef.current;
   197→            setHasSigned(pad ? !pad.isEmpty() : false);
   198→          },
   199→        });
   200→        sigPadInitialized.current = true;
   201→
   202→        const handleResize = () => {
   203→          if (!sigPadRef.current) return;
   204→          const ratio = Math.max(window.devicePixelRatio || 1, 1);
   205→          const dataUrl = sigPadRef.current.toData();
   206→          canvas.width = canvas.offsetWidth * ratio;
   207→          canvas.height = canvas.offsetHeight * ratio;
   208→          const ctx = canvas.getContext("2d");
   209→          if (ctx) ctx.scale(ratio, ratio);
   210→          if (dataUrl) sigPadRef.current.fromData(dataUrl);
   211→        };
   212→
   213→        window.addEventListener("resize", handleResize);
   214→        (sigPadRef as Record<string, unknown>)._cleanup = () => {
   215→          window.removeEventListener("resize", handleResize);
   216→          sigPadRef.current?.off();
   217→        };
   218→      });
   219→    }).catch((err) => {
   220→      console.error("Failed to load SignaturePad:", err);
   221→    });
   222→
   223→    return () => {
   224→      destroyed = true;
   225→      const cleanup = (sigPadRef as Record<string, unknown>)._cleanup as (() => void) | undefined;
   226→      if (cleanup) {
   227→        cleanup();
   228→        delete (sigPadRef as Record<string, unknown>)._cleanup;
   229→      }
   230→      sigPadInitialized.current = false;
   231→    };
   232→  }, [allSkillsRated]); // eslint-disable-line react-hooks/exhaustive-deps
   233→
   234→  /* ─── Save rating ────────────────────────────────────────────────── */
   235→  const saveRating = async (
   236→    skillId: number,
   237→    ratingValue: string | null,
   238→    isNa: boolean
   239→  ) => {
   240→    if (!id) return;
   241→    setAutoSaveStatus("saving");
   242→    try {
   243→      const res = await fetch(`/api/candidate/checklists/${id}/rate`, {
   244→        method: "PUT",
   245→        headers: { "Content-Type": "application/json" },
   246→        body: JSON.stringify({ skillId, ratingValue, isNa }),
   247→      });
   248→
   249→      if (res.ok) {
   250→        const resData = await res.json();
   251→        if (resData.rating) {
   252→          setRatings((prev) => ({
   253→            ...prev,
   254→            [skillId]: {
   255→              id: resData.rating.id,
   256→              ratingValue: resData.rating.ratingValue,
   257→              isNa: resData.rating.isNa,
   258→            },
   259→          }));
   260→        }
   261→        if (data) {
   262→          setData({
   263→            ...data,
   264→            completionPct: resData.completionPct ?? data.completionPct,
   265→            ratedSkills: resData.ratedSkills ?? data.ratedSkills,
   266→          });
   267→        }
   268→      }
   269→    } catch {
   270→      // Silent auto-save failure
   271→    } finally {
   272→      setAutoSaveStatus("saved");
   273→      setTimeout(() => setAutoSaveStatus("idle"), 2000);
   274→    }
   275→  };
   276→
   277→  // Debounced save for text inputs
   278→  const debouncedSave = useCallback(
   279→    (() => {
   280→      let timer: ReturnType<typeof setTimeout>;
   281→      return (skillId: number, value: string) => {
   282→        clearTimeout(timer);
   283→        timer = setTimeout(() => saveRating(skillId, value, false), 300);
   284→      };
   285→    })(),
   286→    [id, ratings] // eslint-disable-line react-hooks/exhaustive-deps
   287→  );
   288→
   289→  /* ─── Submit handler ─────────────────────────────────────────────── */
   290→  const handleSubmit = async () => {
   291→    if (!id || !data) return;
   292→
   293→    const unrated = data.skills.filter((s) => !isRatingDone(ratings[s.id]));
   294→
   295→    if (unrated.length > 0) {
   296→      toast.error("Please rate all skills before submitting", {
   297→        description: `${unrated.length} skill${unrated.length > 1 ? "s" : ""} still need ratings`,
   298→      });
   299→      return;
   300→    }
   301→
   302→    if (!candidateNameSigned.trim()) {
   303→      toast.error("Please type your full legal name.");
   304→      return;
   305→    }
   306→
   307→    if (!hasSigned || sigPadRef.current?.isEmpty()) {
   308→      toast.error("Please draw your signature.");
   309→      return;
   310→    }
   311→
   312→    const signatureBase64 = sigPadRef.current?.toDataURL("image/png") || "";
   313→
   314→    setIsSubmitting(true);
   315→    try {
   316→      const res = await fetch(`/api/candidate/checklists/${id}/submit`, {
   317→        method: "POST",
   318→        headers: { "Content-Type": "application/json" },
   319→        body: JSON.stringify({
   320→          candidateNameSigned: candidateNameSigned.trim(),
   321→          signatureBase64,
   322→        }),
   323→      });
   324→
   325→      if (!res.ok) {
   326→        const resData = await res.json();
   327→        toast.error("Submit failed", { description: resData.error });
   328→        return;
   329→      }
   330→
   331→      router.push(`/checklists/${id}/thank-you`);
   332→    } catch {
   333→      toast.error("Failed to submit checklist");
   334→    } finally {
   335→      setIsSubmitting(false);
   336→    }
   337→  };
   338→
   339→  /* ─── Group skills by category ──────────────────────────────────── */
   340→  const groupedSkills =
   341→    data?.skills.reduce<Record<string, SkillItem[]>>((acc, skill) => {
   342→      const cat = skill.category || "General";
   343→      if (!acc[cat]) acc[cat] = [];
   344→      acc[cat].push(skill);
   345→      return acc;
   346→    }, {}) ?? {};
   347→
   348→  const totalSkills = data?.totalSkills ?? 0;
   349→  const ratedSkills = data?.ratedSkills ?? 0;
   350→  const completionPct = data?.completionPct ?? 0;
   351→
   352→  /* ─── Loading ─────────────────────────────────────────────────── */
   353→  if (isLoading) {
   354→    return (
   355→      <div className="max-w-[820px] mx-auto space-y-6 p-6">
   356→        <Skeleton className="h-10 w-full" />
   357→        <Skeleton className="h-24 w-full" />
   358→        {[1, 2, 3, 4].map((i) => (
   359→          <Skeleton key={i} className="h-40 w-full" />
   360→        ))}
   361→      </div>
   362→    );
   363→  }
   364→
   365→  if (!data) return null;
   366→
   367→  // Already submitted
   368→  if (
   369→    data.candidateResponse?.status === "submitted" ||
   370→    data.checklistRequest.status === "completed"
   371→  ) {
   372→    return (
   373→      <div className="max-w-[820px] mx-auto p-6">
   374→        <Card>
   375→          <CardContent className="p-8 text-center">
   376→            <div className="size-14 rounded-full bg-[#DCFCE7] flex items-center justify-center mx-auto mb-4">
   377→              <CheckCircle2 className="size-7 text-[#166534]" />
   378→            </div>
   379→            <h3 className="text-lg font-semibold mb-2">Checklist Already Submitted</h3>
   380→            <p className="text-sm text-[#6B7280]">
   381→              You have already completed this checklist. Thank you!
   382→            </p>
   383→            <Link href="/checklists">
   384→              <Button variant="outline" className="mt-4 gap-2">
   385→                ← Back to Checklists
   386→              </Button>
   387→            </Link>
   388→          </CardContent>
   389→        </Card>
   390→      </div>
   391→    );
   392→  }
   393→
   394→  return (
   395→    <div className="max-w-[820px] mx-auto pb-28">
   396→      {/* ── Sticky Top Bar ──────────────────────────────────────────── */}
   397→      <div className="sticky top-0 z-50 bg-white border-b border-[#E5E7EB]">
   398→        {/* Progress bar (very top) */}
   399→        <div className="h-[5px] bg-[#F3F4F6]">
   400→          <div
   401→            className="h-full bg-[#166534] transition-all duration-500 ease-out"
   402→            style={{ width: `${completionPct}%` }}
   403→          />
   404→        </div>
   405→
   406→        <div className="flex items-center justify-between px-6 py-3">
   407→          <div className="flex items-center gap-3 min-w-0">
   408→            <Link href="/checklists">
   409→              <Button variant="ghost" size="sm" className="gap-1 shrink-0">
   410→                ← Back
   411→              </Button>
   412→            </Link>
   413→            <div className="min-w-0">
   414→              <h1 className="text-sm font-semibold truncate">{data.template.name}</h1>
   415→              <p className="text-xs text-[#6B7280] truncate">
   416→                {data.template.profession}
   417→                {data.template.specialty ? ` — ${data.template.specialty}` : ""}
   418→                {data.client.organizationName ? ` · ${data.client.organizationName}` : ""}
   419→              </p>
   420→            </div>
   421→          </div>
   422→          <div className="flex items-center gap-3 shrink-0">
   423→            {autoSaveStatus === "saving" && (
   424→              <span className="text-xs text-[#6B7280] flex items-center gap-1">
   425→                <Loader2 className="size-3 animate-spin" /> Saving...
   426→              </span>
   427→            )}
   428→            {autoSaveStatus === "saved" && (
   429→              <span className="text-xs text-[#166534] font-medium">Saved ✓</span>
   430→            )}
   431→            <Badge
   432→              variant={completionPct === 100 ? "default" : "secondary"}
   433→              className="text-xs"
   434→            >
   435→              {completionPct}% Complete
   436→            </Badge>
   437→          </div>
   438→        </div>
   439→      </div>
   440→
   441→      <div className="p-6 space-y-6">
   442→        {/* ── Step Indicator (2 steps) ─────────────────────────────── */}
   443→        <div className="flex items-center gap-2 text-xs font-medium">
   444→          <div className={`flex items-center gap-1.5 ${allSkillsRated ? "text-[#166534]" : "text-[#166534]"}`}>
   445→            <div className={`size-6 rounded-full flex items-center justify-center text-[10px] font-bold ${allSkillsRated ? "bg-[#166534] text-white" : "border-2 border-[#166534] text-[#166534]"}`}>
   446→              {allSkillsRated ? "✓" : "1"}
   447→            </div>
   448→            Skills Assessment
   449→          </div>
   450→          <ChevronRight className="size-3 text-[#9CA3AF]" />
   451→          <div className={`flex items-center gap-1.5 ${allSkillsRated ? "text-[#166534]" : "text-[#9CA3AF]"}`}>
   452→            <div className={`size-6 rounded-full flex items-center justify-center text-[10px] font-bold ${allSkillsRated ? "border-2 border-[#166534] text-[#166534]" : "border-2 border-[#E5E7EB] text-[#9CA3AF]"}`}>
   453→              2
   454→            </div>
   455→            Sign & Submit
   456→          </div>
   457→        </div>
   458→
   459→        {/* ── Section 1: Skills Assessment ────────────────────────── */}
   460→        {data.skills.length === 0 ? (
   461→          <Card>
   462→            <CardContent className="p-8 text-center">
   463→              <Lock className="size-12 text-[#9CA3AF] mx-auto mb-4" />
   464→              <h3 className="text-lg font-medium">No skills found</h3>
   465→              <p className="text-sm text-[#6B7280] mt-1">
   466→                This checklist template has no skills configured yet.
   467→              </p>
   468→            </CardContent>
   469→          </Card>
   470→        ) : (
   471→          <div className="space-y-6">
   472→            {Object.entries(groupedSkills).map(([category, categorySkills]) => {
   473→              const ratedInCat = categorySkills.filter((s) => isRatingDone(ratings[s.id])).length;
   474→
   475→              return (
   476→                <div key={category}>
   477→                  {/* Category header */}
   478→                  <div className="bg-white border border-[#E5E7EB] border-l-4 border-l-[#166534] rounded-r-xl py-3.5 px-5 flex items-center justify-between mb-3">
   479→                    <span className="font-semibold text-sm">{category}</span>
   480→                    <span className="text-xs text-[#9CA3AF]">
   481→                      {ratedInCat}/{categorySkills.length} rated
   482→                    </span>
   483→                  </div>
   484→
   485→                  {/* Skills in this category */}
   486→                  <div className="space-y-3">
   487→                    {categorySkills.map((skill) => {
   488→                      const rating = ratings[skill.id];
   489→                      const currentValue = rating?.ratingValue ?? null;
   490→                      const isNa = rating?.isNa ?? false;
   491→                      const isRated = isRatingDone(rating);
   492→
   493→                      return (
   494→                        <div
   495→                          key={skill.id}
   496→                          className={`bg-white border rounded-xl p-4 sm:p-5 transition-all ${
   497→                            isRated
   498→                              ? "border-[#166534]/20 shadow-sm"
   499→                              : "border-[#E5E7EB] hover:border-[#D1D5DB]"
   500→                          }`}
   501→                        >
   502→                          <div className="flex flex-col gap-3">
   503→                            {/* Skill name + N/A toggle */}
   504→                            <div className="flex items-start gap-3">
   505→                              <p className="text-sm font-medium flex-1">{skill.skillName}</p>
   506→                              {skill.hasNaOption && (
   507→                                <button
   508→                                  type="button"
   509→                                  onClick={() => saveRating(skill.id, null, !isNa)}
   510→                                  className={`text-xs px-2.5 py-1 rounded-full border transition-all shrink-0 ${
   511→                                    isNa
   512→                                      ? "bg-[#0D9488]/10 border-[#0D9488] text-[#0D9488] font-medium"
   513→                                      : "border-[#E5E7EB] text-[#9CA3AF] hover:border-[#0D9488]"
   514→                                  }`}
   515→                                >
   516→                                  N/A
   517→                                </button>
   518→                              )}
   519→                            </div>
   520→
   521→                            {/* Rating 1-5 */}
   522→                            {skill.questionType === "rating_1_5" && !isNa && (
   523→                              <div className="flex flex-wrap gap-2">
   524→                                {[1, 2, 3, 4, 5].map((val) => {
   525→                                  const isSelected = currentValue === String(val);
   526→                                  const color = RATING_COLORS[val];
   527→                                  return (
   528→                                    <button
   529→                                      key={val}
   530→                                      type="button"
   531→                                      onClick={() => saveRating(skill.id, String(val), false)}
   532→                                      className={`flex flex-col items-center px-3 py-2 rounded-lg border text-sm transition-all ${
   533→                                        isSelected
   534→                                          ? "border-2 font-semibold scale-105"
   535→                                          : "border-[#E5E7EB] hover:border-[#D1D5DB]"
   536→                                      }`}
   537→                                      style={
   538→                                        isSelected
   539→                                          ? { borderColor: color.border, backgroundColor: color.bg, color: color.text }
   540→                                          : undefined
   541→                                      }
   542→                                    >
   543→                                      <span className="text-base font-bold">{val}</span>
   544→                                      <span className="text-[10px] opacity-70 mt-0.5">{RATING_LABELS[val]}</span>
   545→                                    </button>
   546→                                  );
   547→                                })}
   548→                              </div>
   549→                            )}
   550→
   551→                            {/* Yes/No */}
   552→                            {skill.questionType === "yes_no" && !isNa && (
   553→                              <div className="flex gap-2">
   554→                                <button
   555→                                  type="button"
   556→                                  onClick={() => saveRating(skill.id, "yes", false)}
   557→                                  className={`px-4 py-2 rounded-lg border text-sm font-medium transition-all ${
