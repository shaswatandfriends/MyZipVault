    54→interface Announcement {
    55→  id: number;

    62→// ─── Helpers ────────────────────────────────────────────────────────
    63→function getTargetBadge(targetRole: string) {
    64→  switch (targetRole) {
    65→    case "all":
    66→      return <Badge className="bg-teal-100 text-teal-800 border-teal-200 hover:bg-teal-100">All</Badge>;
    67→    case "candidate":
    68→      return <Badge className="bg-emerald-100 text-emerald-800 border-emerald-200 hover:bg-emerald-100">Candidates</Badge>;
    69→    case "client_recruiter":
    70→      return <Badge className="bg-amber-100 text-amber-800 border-amber-200 hover:bg-amber-100">Recruiters</Badge>;
    71→    default:
    72→      return <Badge variant="outline">{targetRole}</Badge>;
    73→  }
    74→}
    75→
    76→function formatDate(dateStr: string): string {
    77→  return new Date(dateStr).toLocaleDateString("en-US", {
    78→    month: "short",
    79→    day: "numeric",
    80→    year: "numeric",
    81→  });
    82→}

   117→// ─── Main Component ─────────────────────────────────────────────────
   118→export default function SuperadminAnnouncementsPage() {
   119→  const [announcements, setAnnouncements] = useState<Announcement[]>([]);
   120→  const [isLoading, setIsLoading] = useState(true);
   121→  const [createOpen, setCreateOpen] = useState(false);
   122→  const [editId, setEditId] = useState<number | null>(null);
   123→
   124→  // Form state
   125→  const [formMessage, setFormMessage] = useState("");
   126→  const [formTargetRole, setFormTargetRole] = useState("all");
   127→  const [formIsActive, setFormIsActive] = useState(false);
   128→  const [saving, setSaving] = useState(false);
   129→
   130→  // Delete confirmation
   131→  const [deleteId, setDeleteId] = useState<number | null>(null);
   132→
   133→  // Email section
   134→  const [emailSegment, setEmailSegment] = useState("all_candidates");
   135→  const [emailTemplate, setEmailTemplate] = useState("");
   136→  const [emailAnnouncementId, setEmailAnnouncementId] = useState<number | null>(null);
   137→  const [sendingCampaign, setSendingCampaign] = useState(false);
   138→  const [previewOpen, setPreviewOpen] = useState(false);
   139→  const [campaignResult, setCampaignResult] = useState<{
   140→    sentCount: number;
   141→    failedCount: number;
   142→    totalTargets: number;
   143→    notificationsCreated: number;
   144→  } | null>(null);
   145→
   146→  const fetchAnnouncements = useCallback(async () => {
   147→    try {
   148→      setIsLoading(true);
   149→      const res = await fetch("/api/superadmin/announcements");
   150→      if (!res.ok) {
   151→        const body = await res.json().catch(() => ({}));
   152→        throw new Error(body.error || "Failed to fetch announcements");
   153→      }
   154→      const json = await res.json();
   155→      setAnnouncements(json.announcements);
   156→    } catch (err) {
   157→      const message = err instanceof Error ? err.message : "Something went wrong";
   158→      toast.error("Failed to load announcements", { description: message });
   159→    } finally {
   160→      setIsLoading(false);
   161→    }
   162→  }, []);
   163→
   164→  useEffect(() => {
   165→    fetchAnnouncements();
   166→  }, [fetchAnnouncements]);
   167→
   168→  const resetForm = () => {
   169→    setFormMessage("");
   170→    setFormTargetRole("all");
   171→    setFormIsActive(false);
   172→    setEditId(null);
   173→  };
   174→
   175→  const openCreate = () => {
   176→    resetForm();
   177→    setCreateOpen(true);
   178→  };
   179→
   180→  const openEdit = (a: Announcement) => {
   181→    setFormMessage(a.message);
   182→    setFormTargetRole(a.targetRole);
   183→    setFormIsActive(a.isActive);
   184→    setEditId(a.id);
   185→    setCreateOpen(true);
   186→  };
   187→
   188→  const handleSave = async () => {
   189→    if (!formMessage.trim()) {
   190→      toast.error("Message is required");
   191→      return;
   192→    }
   193→    try {
   194→      setSaving(true);
   195→      const payload: Record<string, unknown> = {
   196→        action: editId ? "update" : "create",
   197→        message: formMessage,
   198→        targetRole: formTargetRole,
   199→        isActive: formIsActive,
   200→      };
   201→      if (editId) payload.id = editId;
   202→
   203→      const res = await fetch("/api/superadmin/announcements", {
   204→        method: "POST",
   205→        headers: { "Content-Type": "application/json" },
   206→        body: JSON.stringify(payload),
   207→      });
   208→      const json = await res.json();
   209→      if (!res.ok) throw new Error(json.error || "Failed to save");
   210→      toast.success(editId ? "Announcement updated" : "Announcement created");
   211→      setCreateOpen(false);
   212→      resetForm();
   213→      fetchAnnouncements();
   214→    } catch (err) {
   215→      const message = err instanceof Error ? err.message : "Failed to save";
   216→      toast.error("Save failed", { description: message });
   217→    } finally {
   218→      setSaving(false);
   219→    }
   220→  };
   221→
   222→  const handleToggle = async (id: number, isActive: boolean) => {
   223→    try {
   224→      const res = await fetch("/api/superadmin/announcements", {
   225→        method: "POST",
   226→        headers: { "Content-Type": "application/json" },
   227→        body: JSON.stringify({ action: "toggle", id, isActive }),
   228→      });
   229→      const json = await res.json();
   230→      if (!res.ok) throw new Error(json.error || "Failed to toggle");
   231→      toast.success(isActive ? "Announcement enabled" : "Announcement disabled");
   232→      fetchAnnouncements();
   233→    } catch (err) {
   234→      const message = err instanceof Error ? err.message : "Failed to toggle";
   235→      toast.error("Toggle failed", { description: message });
   236→    }
   237→  };
   238→
   239→  const handleDelete = async () => {
   240→    if (!deleteId) return;
   241→    try {
   242→      const res = await fetch("/api/superadmin/announcements", {
   243→        method: "POST",
   244→        headers: { "Content-Type": "application/json" },
   245→        body: JSON.stringify({ action: "delete", id: deleteId }),
   246→      });
   247→      const json = await res.json();
   248→      if (!res.ok) throw new Error(json.error || "Failed to delete");
   249→      toast.success("Announcement deleted");
   250→      setDeleteId(null);
   251→      fetchAnnouncements();
   252→    } catch (err) {
   253→      const message = err instanceof Error ? err.message : "Failed to delete";
   254→      toast.error("Delete failed", { description: message });
   255→    }
   256→  };
   257→
   258→  const handleSendCampaign = async () => {
   259→    if (!emailTemplate) {
   260→      toast.error("Please select an email template");
   261→      return;
   262→    }
   263→    try {
   264→      setSendingCampaign(true);
   265→      setCampaignResult(null);
   266→
   267→      const res = await fetch("/api/superadmin/announcements", {
   268→        method: "POST",
   269→        headers: { "Content-Type": "application/json" },
   270→        body: JSON.stringify({
   271→          action: "send_campaign",
   272→          announcementId: emailAnnouncementId,
   273→          targetRoles: [emailSegment],
   274→          sendEmail: true,
   275→          emailTemplate,
   276→        }),
   277→      });
   278→
   279→      const json = await res.json();
   280→      if (!res.ok) throw new Error(json.error || "Failed to send campaign");
   281→
   282→      setCampaignResult({
   283→        sentCount: json.sentCount,
   284→        failedCount: json.failedCount,
   285→        totalTargets: json.totalTargets,
   286→        notificationsCreated: json.notificationsCreated,
   287→      });
   288→
   289→      if (json.failedCount > 0) {
   290→        toast.warning("Campaign partially sent", {
   291→          description: `${json.sentCount} emails sent, ${json.failedCount} failed. ${json.notificationsCreated} notifications created.`,
   292→          duration: 6000,
   293→        });
   294→      } else {
   295→        toast.success("Email campaign sent!", {
   296→          description: `${json.sentCount} emails sent. ${json.notificationsCreated} in-app notifications created.`,
   297→          duration: 5000,
   298→        });
   299→      }
   300→    } catch (err) {
   301→      const message = err instanceof Error ? err.message : "Failed to send campaign";
   302→      toast.error("Campaign failed", { description: message });
   303→    } finally {
   304→      setSendingCampaign(false);
   305→    }
   306→  };
   307→
   308→  const activeCount = announcements.filter((a) => a.isActive).length;
   309→
   310→  return (
   311→    <div className="space-y-6">
   312→      <PageHeader
   313→        title="Announcements & Communication"
   314→        description="Send in-app banners and targeted emails to user segments."
   315→        actions={
   316→          <Button
   317→            className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2"
   318→            onClick={openCreate}
   319→          >
   320→            <Plus className="size-4" />
   321→            Create Announcement
   322→          </Button>
   323→        }
   324→      />
   325→
   326→      {/* ── Stats ─────────────────────────────────────────────────── */}
   327→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   328→        <Card className="hover:shadow-md transition-shadow">
   329→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   330→            <CardTitle className="text-sm font-medium">Total Announcements</CardTitle>
   331→            <div className="size-8 rounded-lg bg-teal-50 flex items-center justify-center">
   332→              <Megaphone className="size-4 text-teal-600" />
   333→            </div>
   334→          </CardHeader>
   335→          <CardContent>
   336→            <div className="text-2xl font-bold">{announcements.length}</div>
   337→          </CardContent>
   338→        </Card>
   339→        <Card className="hover:shadow-md transition-shadow">
   340→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   341→            <CardTitle className="text-sm font-medium">Active</CardTitle>
   342→            <div className="size-8 rounded-lg bg-emerald-50 flex items-center justify-center">
   343→              <ToggleLeft className="size-4 text-emerald-600" />
   344→            </div>
   345→          </CardHeader>
   346→          <CardContent>
   347→            <div className="text-2xl font-bold text-emerald-700">{activeCount}</div>
   348→          </CardContent>
   349→        </Card>
   350→        <Card className="hover:shadow-md transition-shadow">
   351→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   352→            <CardTitle className="text-sm font-medium">Inactive</CardTitle>
   353→            <div className="size-8 rounded-lg bg-muted flex items-center justify-center">
   354→              <Megaphone className="size-4 text-muted-foreground" />
   355→            </div>
   356→          </CardHeader>
   357→          <CardContent>
   358→            <div className="text-2xl font-bold">{announcements.length - activeCount}</div>
   359→          </CardContent>
   360→        </Card>
   361→      </div>
   362→
   363→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   364→        {/* ── Announcements List ──────────────────────────────────── */}
   365→        <div className="lg:col-span-2">
   366→          <Card>
   367→            <CardHeader>
   368→              <CardTitle className="text-base">All Announcements</CardTitle>
   369→            </CardHeader>
   370→            <CardContent>
   371→              {isLoading ? (
   372→                <AnnouncementSkeleton />
   373→              ) : announcements.length === 0 ? (
   374→                <div className="flex flex-col items-center justify-center py-12 text-center">
   375→                  <Megaphone className="size-12 text-muted-foreground mb-4" />
   376→                  <h3 className="text-lg font-semibold mb-1">No announcements</h3>
   377→                  <p className="text-sm text-muted-foreground">Create your first announcement to get started.</p>
   378→                </div>
   379→              ) : (
   380→                <div className="max-h-[28rem] overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border space-y-0">
   381→                  {announcements.map((a) => (
   382→                    <div
   383→                      key={a.id}
   384→                      className="flex items-start gap-3 py-3 border-b last:border-0"
   385→                    >
   386→                      <div className="flex-1 min-w-0">
   387→                        <p className="text-sm font-medium">{a.message}</p>
   388→                        <div className="flex items-center gap-2 mt-1">
   389→                          {getTargetBadge(a.targetRole)}
   390→                          <span className="text-xs text-muted-foreground">
   391→                            {formatDate(a.createdAt)}
   392→                          </span>
   393→                        </div>
   394→                      </div>
   395→                      <div className="flex items-center gap-2 shrink-0">
   396→                        <Switch
   397→                          checked={a.isActive}
   398→                          onCheckedChange={(checked) => handleToggle(a.id, checked)}
   399→                        />
   400→                        <Button
   401→                          variant="ghost"
   402→                          size="sm"
   403→                          className="size-8 p-0"
   404→                          onClick={() => openEdit(a)}
   405→                        >
   406→                          <Eye className="size-4" />
   407→                        </Button>
   408→                        <Button
   409→                          variant="ghost"
   410→                          size="sm"
   411→                          className="size-8 p-0 text-red-600 hover:text-red-700"
   412→                          onClick={() => setDeleteId(a.id)}
   413→                        >
   414→                          <Trash2 className="size-4" />
   415→                        </Button>
   416→                      </div>
   417→                    </div>
   418→                  ))}
   419→                </div>
   420→              )}
   421→            </CardContent>
   422→          </Card>
   423→        </div>
   424→
   425→        {/* ── Send Email Section ──────────────────────────────────── */}
   426→        <Card>
   427→          <CardHeader>
   428→            <div className="flex items-center gap-2">
   429→              <div className="size-8 rounded-lg bg-teal-50 flex items-center justify-center">
   430→                <Mail className="size-4 text-teal-600" />
   431→              </div>
   432→              <div>
   433→                <CardTitle className="text-base">Send Email</CardTitle>
   434→                <CardDescription>Targeted email campaigns</CardDescription>
   435→              </div>
   436→            </div>
   437→          </CardHeader>
   438→          <CardContent className="space-y-4">
   439→            <div className="space-y-2">
   440→              <Label className="text-xs font-medium">Target Segment</Label>
   441→              <Select value={emailSegment} onValueChange={setEmailSegment}>
   442→                <SelectTrigger>
   443→                  <SelectValue />
   444→                </SelectTrigger>
   445→                <SelectContent>
   446→                  <SelectItem value="all_candidates">All Candidates</SelectItem>
   447→                  <SelectItem value="expiring_credentials">Candidates with Expiring Credentials</SelectItem>
   448→                  <SelectItem value="all_recruiters">All Recruiters</SelectItem>
   449→                  <SelectItem value="inactive_users">Inactive Users (30d+)</SelectItem>
   450→                </SelectContent>
   451→              </Select>
   452→            </div>
   453→
   454→            <div className="space-y-2">
   455→              <Label className="text-xs font-medium">Email Template</Label>
   456→              <Select value={emailTemplate} onValueChange={setEmailTemplate}>
   457→                <SelectTrigger>
   458→                  <SelectValue placeholder="Select template…" />
   459→                </SelectTrigger>
   460→                <SelectContent>
   461→                  <SelectItem value="credential_expiry_reminder">Credential Expiry Reminder</SelectItem>
   462→                  <SelectItem value="profile_completion">Profile Completion Nudge</SelectItem>
   463→                  <SelectItem value="new_features">New Features Announcement</SelectItem>
   464→                  <SelectItem value="monthly_digest">Monthly Digest</SelectItem>
   465→                </SelectContent>
   466→              </Select>
   467→            </div>
   468→
   469→            <div className="space-y-2">
   470→              <Label className="text-xs font-medium">Attach Announcement (Optional)</Label>
   471→              <Select
   472→                value={emailAnnouncementId ? String(emailAnnouncementId) : "none"}
   473→                onValueChange={(val) => setEmailAnnouncementId(val === "none" ? null : parseInt(val, 10))}
   474→              >
   475→                <SelectTrigger>
   476→                  <SelectValue placeholder="None" />
   477→                </SelectTrigger>
   478→                <SelectContent>
   479→                  <SelectItem value="none">No Announcement</SelectItem>
   480→                  {announcements.filter((a) => a.isActive).map((a) => (
   481→                    <SelectItem key={a.id} value={String(a.id)}>
   482→                      {a.message.slice(0, 50)}{a.message.length > 50 ? "…" : ""}
   483→                    </SelectItem>
   484→                  ))}
   485→                </SelectContent>
   486→              </Select>
   487→            </div>
   488→
   489→            <Separator />
   490→
   491→            <div className="rounded-lg border border-dashed p-4 text-center">
   492→              <Mail className="size-8 text-muted-foreground mx-auto mb-2" />
   493→              <p className="text-sm font-medium">Email Preview</p>
   494→              <p className="text-xs text-muted-foreground mt-1">
   495→                {segmentLabels[emailSegment] || emailSegment}
   496→              </p>
   497→              <p className="text-xs text-muted-foreground mt-1">
   498→                Template: {emailTemplate ? emailTemplate.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase()) : "None selected"}
   499→              </p>
   500→              {emailAnnouncementId && (
   501→                <p className="text-xs text-teal-600 mt-1">
   502→                  + Announcement attached
   503→                </p>
   504→              )}
   505→            </div>
   506→
   507→            {/* Campaign Result */}
   508→            {campaignResult && (
   509→              <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-3">
   510→                <div className="flex items-center gap-2 text-sm font-medium text-emerald-800">
   511→                  <CheckCircle2 className="size-4" />
   512→                  Campaign Sent
   513→                </div>
   514→                <div className="mt-2 grid grid-cols-2 gap-2 text-xs">
   515→                  <div>
   516→                    <span className="text-emerald-600 font-semibold">{campaignResult.sentCount}</span>
   517→                    <span className="text-emerald-700"> emails sent</span>
   518→                  </div>
   519→                  <div>
   520→                    <span className="text-emerald-600 font-semibold">{campaignResult.notificationsCreated}</span>
   521→                    <span className="text-emerald-700"> notifications</span>
   522→                  </div>
   523→                  {campaignResult.failedCount > 0 && (
   524→                    <div>
   525→                      <span className="text-red-600 font-semibold">{campaignResult.failedCount}</span>
   526→                      <span className="text-red-700"> failed</span>
   527→                    </div>
   528→                  )}
   529→                  <div>
   530→                    <span className="text-emerald-600 font-semibold">{campaignResult.totalTargets}</span>
   531→                    <span className="text-emerald-700"> total targets</span>
   532→                  </div>
   533→                </div>
   534→              </div>
   535→            )}
   536→
   537→            <div className="flex gap-2">
   538→              <Button
   539→                variant="outline"
   540→                size="sm"
   541→                className="flex-1 gap-2"
   542→                disabled={!emailTemplate}
   543→                onClick={() => setPreviewOpen(true)}
   544→              >
   545→                <Eye className="size-4" />
   546→                Preview
   547→              </Button>
   548→              <Button
   549→                className="flex-1 gap-2 bg-emerald-600 hover:bg-emerald-700 text-white"
   550→                size="sm"
   551→                disabled={!emailTemplate || sendingCampaign}
   552→                onClick={handleSendCampaign}
   553→              >
   554→                {sendingCampaign ? (
   555→                  <Loader2 className="size-4 animate-spin" />
   556→                ) : (
   557→                  <Send className="size-4" />
   558→                )}
   559→                {sendingCampaign ? "Sending…" : "Send"}
   560→              </Button>
   561→            </div>
   562→          </CardContent>
   563→        </Card>
   564→      </div>
   565→
   566→      {/* ── Create / Edit Announcement Dialog ─────────────────────── */}
   567→      <Dialog
   568→        open={createOpen}
   569→        onOpenChange={(open) => {
