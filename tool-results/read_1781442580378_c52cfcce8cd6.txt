   234→/* ─── Helper: format time for display ──────────────────────────────── */
   235→function formatTimeDisplay(time: string | null): string {
   236→  if (!time) return "All day";
   237→  const [h, m] = time.split(":").map(Number);
   238→  const period = h >= 12 ? "PM" : "AM";
   239→  const displayH = h === 0 ? 12 : h > 12 ? h - 12 : h;
   240→  return `${displayH}:${m.toString().padStart(2, "0")} ${period}`;
   241→}
