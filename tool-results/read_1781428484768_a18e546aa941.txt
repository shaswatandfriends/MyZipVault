    18→async function sendVaultSignEmail({ to, toName, subject, htmlContent }: VaultSignEmailParams): Promise<boolean> {
    19→  if (BREVO_API_KEY) {
    20→    try {
    21→      const response = await fetch(BREVO_API_URL, {
    22→        method: "POST",
    23→        headers: {
    24→          "api-key": BREVO_API_KEY,
    25→          "Content-Type": "application/json",
    26→        },
    27→        body: JSON.stringify({
    28→          sender: { email: BREVO_SENDER_EMAIL, name: "VaultSign by MyZipVault" },
    29→          to: [{ email: to, name: toName || to.split("@")[0] }],
    30→          subject,
    31→          htmlContent,
    32→        }),
    33→      });
    34→
    35→      if (response.ok) {
    36→        console.log(`[VAULTSIGN-EMAIL] Sent to ${to}: ${subject}`);
    37→        return true;
    38→      } else {
    39→        const error = await response.text();
    40→        console.error(`[VAULTSIGN-EMAIL] Brevo API error: ${error}`);
    41→      }
    42→    } catch (error) {
    43→      console.error("[VAULTSIGN-EMAIL] Brevo API call failed:", error);
    44→    }
    45→  }
    46→
    47→  // Fallback: log to console
    48→  console.log(`[VAULTSIGN-EMAIL] To: ${to}`);
    49→  console.log(`[VAULTSIGN-EMAIL] Subject: ${subject}`);
    50→  console.log(`[VAULTSIGN-EMAIL] Body: ${htmlContent.substring(0, 300)}...`);
    51→  return true;
    52→}

   151→// ─── Expiry Urgency Helper ───────────────────────────────────────────────────
   152→
   153→function expiryBlock(expiryDate?: string): string {
   154→  if (!expiryDate) return "";
   155→
   156→  let daysLeft = Infinity;
   157→  try {
   158→    const expiry = new Date(expiryDate);
   159→    const now = new Date();
   160→    daysLeft = Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   161→  } catch {
   162→    // keep Infinity
   163→  }
   164→
   165→  const isUrgent = daysLeft <= 3;
   166→  const isWarning = daysLeft <= 7 && daysLeft > 3;
   167→
   168→  const bg = isUrgent ? "#FEF2F2" : isWarning ? "#FFFBEB" : "#F8F7F4";
   169→  const border = isUrgent ? "#DC2626" : isWarning ? "#D97706" : "#E5E7EB";
   170→  const color = isUrgent ? "#DC2626" : isWarning ? "#D97706" : "#6B7280";
   171→  const icon = isUrgent ? "&#x26A0;&#xFE0F;" : isWarning ? "&#x23F0;" : "&#x1F4C5;";
   172→  const label = isUrgent
   173→    ? `Expires in ${daysLeft} day${daysLeft === 1 ? "" : "s"} — Action required!`
   174→    : isWarning
   175→      ? `Expires in ${daysLeft} days`
   176→      : `Expires on ${expiryDate}`;
   177→
   178→  return `
   179→  <table role="presentation" width="100%" cellspacing="0" cellpadding="0" style="margin:16px 0;">
   180→    <tr>
   181→      <td style="background-color:${bg}; border:1px solid ${border}; border-radius:8px; padding:12px 16px; text-align:center;">
   182→        <span style="font-size:13px; color:${color}; font-family:${FONT_STACK};">${icon} ${label}</span>
   183→      </td>
   184→    </tr>
   185→  </table>`;
   186→}
