   532→// ─── Main Component ─────────────────────────────────────────────────
   533→export default function LandingPageEditorPage() {
   534→  const [data, setData] = useState<LandingPageData>(defaultData);
   535→  const [previewMode, setPreviewMode] = useState<"desktop" | "mobile">(
   536→    "desktop"
   537→  );
   538→  const [saving, setSaving] = useState(false);
   539→  const [loading, setLoading] = useState(true);
   540→  const [lastSaved, setLastSaved] = useState<string | null>(null);
   541→  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);
   542→  const [loadedFromServer, setLoadedFromServer] = useState(false);
   543→
   544→  // ── Load content from API on mount ──
   545→  const loadContent = useCallback(async () => {
   546→    try {
   547→      setLoading(true);
   548→      const res = await fetch("/api/superadmin/landing-page");
   549→      if (res.ok) {
   550→        const json = await res.json();
   551→        setData(json as LandingPageData);
   552→        setLoadedFromServer(true);
   553→        setHasUnsavedChanges(false);
   554→      }
   555→    } catch {
   556→      // Silently fail — will use default data
   557→    } finally {
   558→      setLoading(false);
   559→    }
   560→  }, []);
   561→
   562→  useEffect(() => {
   563→    loadContent();
   564→  }, [loadContent]);
   565→
   566→  // ── Hero helpers ──
   567→  const updateHero = (key: keyof HeroContent, value: string) => {
   568→    setData((prev) => ({ ...prev, hero: { ...prev.hero, [key]: value } }));
   569→    setHasUnsavedChanges(true);
   570→  };
   571→
   572→  // ── Color helpers ──
   573→  const updateColor = (key: keyof ColorSettings, value: string) => {
   574→    setData((prev) => ({ ...prev, colors: { ...prev.colors, [key]: value } }));
   575→    setHasUnsavedChanges(true);
   576→  };
   577→
   578→  // ── Feature card helpers ──
   579→  const updateFeatureCard = (
   580→    index: number,
   581→    key: keyof FeatureCard,
   582→    value: string
   583→  ) => {
   584→    setData((prev) => {
   585→      const cards = [...prev.featureCards];
   586→      cards[index] = { ...cards[index], [key]: value };
   587→      return { ...prev, featureCards: cards };
   588→    });
   589→    setHasUnsavedChanges(true);
   590→  };
   591→
   592→  // ── Privacy item helpers ──
   593→  const updatePrivacyItem = (
   594→    index: number,
   595→    key: keyof PrivacyItem,
   596→    value: string
   597→  ) => {
   598→    setData((prev) => {
   599→      const items = [...prev.privacySection];
   600→      items[index] = { ...items[index], [key]: value };
   601→      return { ...prev, privacySection: items };
   602→    });
   603→    setHasUnsavedChanges(true);
   604→  };
   605→
   606→  // ── How It Works helpers ──
   607→  const updateHowItWorksStep = (
   608→    index: number,
   609→    key: keyof HowItWorksStep,
   610→    value: string
   611→  ) => {
   612→    setData((prev) => {
   613→      const steps = [...prev.howItWorks];
   614→      steps[index] = { ...steps[index], [key]: value };
   615→      return { ...prev, howItWorks: steps };
   616→    });
   617→    setHasUnsavedChanges(true);
   618→  };
   619→
   620→  // ── Footer helpers ──
   621→  const updateFooter = (key: keyof FooterContent, value: string) => {
