   102→// ---------- AI Assist Button Component ----------
   103→function AiAssistButton({
   104→  label,
   105→  action,
   106→  context,
   107→  currentContent,
   108→  onResult,
   109→}: {
   110→  label: string;
   111→  action: string;
   112→  context?: Record<string, unknown>;
   113→  currentContent?: string;
   114→  onResult: (result: string) => void;
   115→}) {
   116→  const [isLoading, setIsLoading] = useState(false);
   117→
   118→  const handleClick = async () => {
   119→    setIsLoading(true);
   120→    try {
   121→      // Build system and user prompts (same logic as server route)
   122→      let systemPrompt = "";
   123→      let userPrompt = "";
   124→
   125→      switch (action) {
   126→        case "generate_summary": {
   127→          systemPrompt = `You are a professional resume writer specializing in healthcare staffing. Write a compelling professional summary for a healthcare professional's resume. The summary should be concise (2-4 sentences), highlight key qualifications, and be tailored to healthcare positions. Return ONLY the summary text, no additional commentary.`;
   128→          userPrompt = context
   129→            ? `Generate a professional summary based on this information:\n\n${JSON.stringify(context, null, 2)}`
   130→            : "Generate a professional summary for an experienced healthcare professional (nurse/therapist/technician).";
   131→          break;
   132→        }
   133→        case "improve_summary": {
   134→          systemPrompt = `You are a professional resume writer specializing in healthcare staffing. Improve and enhance the given professional summary to make it more compelling, impactful, and tailored for healthcare positions. Keep it concise (2-4 sentences). Return ONLY the improved summary text, no additional commentary.`;
   135→          userPrompt = `Improve this professional summary:\n\n"${currentContent}"`;
   136→          break;
   137→        }
   138→        case "improve_experience": {
   139→          systemPrompt = `You are a professional resume writer specializing in healthcare staffing. Improve and enhance the given work experience description to make it more impactful, using strong action verbs and quantifiable achievements where possible. Tailor it for healthcare positions. Return ONLY the improved description text, no additional commentary.`;
   140→          userPrompt = `Improve this work experience description for a healthcare position:\n\n"${currentContent}"\n\nContext: ${context ? JSON.stringify(context) : "Healthcare professional"}`;
   141→          break;
   142→        }
   143→        case "suggest_skills": {
   144→          systemPrompt = `You are a healthcare staffing expert. Based on the provided context, suggest relevant healthcare skills that the candidate should include in their resume. Return a JSON array of objects with "skill" (string) and "proficiency" (one of: Beginner, Intermediate, Advanced, Expert) fields. Return ONLY the JSON array, no additional text.`;
   145→          userPrompt = context
   146→            ? `Suggest healthcare skills for this professional:\n\n${JSON.stringify(context, null, 2)}`
   147→            : "Suggest common healthcare skills for an experienced nurse or healthcare professional.";
   148→          break;
   149→        }
   150→        case "suggest_certifications": {
   151→          systemPrompt = `You are a healthcare staffing expert. Based on the provided context, suggest relevant healthcare certifications that the candidate should pursue or include in their resume. Return a JSON array of objects with "name" (string), "issuingOrg" (string), and "year" (string) fields. Return ONLY the JSON array, no additional text.`;
   152→          userPrompt = context
   153→            ? `Suggest healthcare certifications for this professional:\n\n${JSON.stringify(context, null, 2)}`
   154→            : "Suggest common healthcare certifications for an experienced nurse or healthcare professional.";
   155→          break;
   156→        }
   157→        case "generate_full_resume": {
   158→          systemPrompt = `You are a professional resume writer specializing in healthcare staffing. Generate complete resume data based on the provided information. Return a JSON object with this exact structure:
   159→{
   160→  "contact": { "fullName": "", "phone": "", "email": "", "address": "" },
   161→  "summary": "",
   162→  "experience": [{ "facility": "", "unit": "", "startDate": "", "endDate": "", "description": "" }],
   163→  "education": [{ "school": "", "degree": "", "year": "" }],
   164→  "certifications": [{ "name": "", "issuingOrg": "", "year": "" }],
   165→  "skills": [{ "skill": "", "proficiency": "Intermediate" }]
   166→}
   167→Return ONLY valid JSON, no additional text or markdown.`;
   168→          userPrompt = `Generate a complete healthcare resume based on this information:\n\n${JSON.stringify(context, null, 2)}`;
   169→          break;
   170→        }
   171→        default: {
   172→          toast.error("Unknown AI action");
   173→          return;
   174→        }
   175→      }
   176→
   177→      // Call AI directly from the browser (bypasses Vercel server network issues)
   178→      const completion = await clientZaiChatCompletion({
   179→        messages: [
   180→          { role: "system", content: systemPrompt },
   181→          { role: "user", content: userPrompt },
   182→        ],
   183→        temperature: 0.7,
   184→        max_tokens: 2000,
   185→      });
   186→
   187→      const result = completion.choices?.[0]?.message?.content || "";
   188→
   189→      if (!result) {
   190→        toast.error("AI could not generate a suggestion");
   191→        return;
   192→      }
   193→
   194→      // For structured actions, try to parse JSON
   195→      if (["suggest_skills", "suggest_certifications", "generate_full_resume"].includes(action)) {
   196→        try {
   197→          const jsonMatch = result.match(/```(?:json)?\s*([\s\S]*?)```/) || [null, result];
   198→          const jsonStr = jsonMatch[1].trim();
   199→          const parsed = JSON.parse(jsonStr);
   200→          onResult(JSON.stringify(parsed));
   201→        } catch {
   202→          onResult(result);
   203→        }
   204→      } else {
   205→        onResult(result);
   206→      }
   207→      toast.success("AI suggestion ready!");
   208→    } catch (err) {
   209→      const errMsg = err instanceof Error ? err.message : "Unknown error";
   210→      toast.error("AI assist failed", { description: errMsg });
   211→    } finally {
   212→      setIsLoading(false);
   213→    }
   214→  };
   215→
   216→  return (
   217→    <Button
   218→      variant="outline"
   219→      size="sm"
   220→      className="gap-1.5 text-violet-600 border-violet-200 hover:bg-violet-50 dark:text-violet-400 dark:border-violet-800 dark:hover:bg-violet-950"
   221→      onClick={handleClick}
   222→      disabled={isLoading}
   223→    >
   224→      {isLoading ? (
   225→        <Loader2 className="size-3.5 animate-spin" />
   226→      ) : (
   227→        <Sparkles className="size-3.5" />
   228→      )}
   229→      {isLoading ? "Generating..." : label}
   230→    </Button>
   231→  );
   232→}

   359→// ---------- AI Chat Panel Component ----------
   360→function AiChatPanel({
   361→  resumeContext,
   362→  onApplySuggestion,
   363→}: {
   364→  resumeContext: ResumeParsedData;
   365→  onApplySuggestion: (action: string, result: unknown) => void;
   366→}) {
   367→  const [messages, setMessages] = useState<ChatMessage[]>([
   368→    {
   369→      role: "assistant",
   370→      content:
   371→        "Hi! I'm your AI resume assistant. I can help you improve your resume, suggest content, and answer questions about healthcare resume best practices. How can I help you today?",
   372→    },
   373→  ]);
   374→  const [input, setInput] = useState("");
   375→  const [isSending, setIsSending] = useState(false);
   376→  const scrollAreaRef = useRef<HTMLDivElement>(null);
   377→  const messagesEndRef = useRef<HTMLDivElement>(null);
   378→
   379→  // Auto-scroll to bottom when messages change using both methods for reliability
   380→  useEffect(() => {
   381→    requestAnimationFrame(() => {
   382→      // Method 1: Direct scrollTop on the scrollable container
   383→      if (scrollAreaRef.current) {
   384→        scrollAreaRef.current.scrollTop = scrollAreaRef.current.scrollHeight;
   385→      }
   386→      // Method 2: scrollIntoView as backup
   387→      messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
   388→    });
   389→  }, [messages, isSending]);
   390→
   391→  const handleSend = async (message?: string) => {
   392→    const trimmed = (message || input).trim();
   393→    if (!trimmed || isSending) return;
   394→
   395→    const userMsg: ChatMessage = { role: "user", content: trimmed };
   396→    setMessages((prev) => [...prev, userMsg]);
   397→    setInput("");
   398→    setIsSending(true);
   399→
   400→    try {
   401→      const systemPrompt = `You are an AI resume assistant for MyZipVault, a healthcare staffing compliance platform. You help candidates improve their resumes, suggest content, and answer questions about resume best practices for healthcare positions. Be helpful, concise, and professional. If asked about something unrelated to resumes or healthcare careers, politely redirect. Format your responses clearly with bullet points or paragraphs as appropriate.`;
   402→
   403→      let userPrompt = trimmed;
   404→      if (resumeContext) {
   405→        userPrompt += `\n\nCandidate's current resume data for context:\n${JSON.stringify(resumeContext, null, 2)}`;
   406→      }
   407→
   408→      // Call AI directly from the browser (bypasses Vercel server network issues)
   409→      const completion = await clientZaiChatCompletion({
   410→        messages: [
   411→          { role: "system", content: systemPrompt },
   412→          { role: "user", content: userPrompt },
   413→        ],
   414→        temperature: 0.7,
   415→        max_tokens: 2000,
   416→      });
   417→
   418→      const resultText = completion.choices?.[0]?.message?.content || "";
   419→      setMessages((prev) => [
   420→        ...prev,
   421→        { role: "assistant", content: resultText || "I couldn't generate a response." },
   422→      ]);
   423→    } catch (err) {
   424→      const errMsg = err instanceof Error ? err.message : "Unknown error";
   425→      setMessages((prev) => [
   426→        ...prev,
   427→        { role: "assistant", content: `Sorry, I encountered an error: ${errMsg}. Please try again later.` },
   428→      ]);
   429→    } finally {
   430→      setIsSending(false);
   431→    }
   432→  };
   433→
   434→  const quickActions = [
   435→    { label: "Generate Summary", message: "Generate a professional summary for my resume" },
   436→    { label: "Suggest Skills", message: "What skills should I add for a healthcare position?" },
   437→    { label: "Improve Experience", message: "Help me improve my work experience descriptions" },
   438→    { label: "Suggest Certifications", message: "What certifications are recommended for healthcare workers?" },
   439→  ];
   440→
   441→  return (
   442→    <div className="flex flex-col h-[calc(100vh-140px)] min-h-[500px] rounded-xl border bg-card text-card-foreground shadow-sm">
   443→      {/* Header */}
   444→      <div className="flex items-center gap-2 px-4 py-3 border-b shrink-0">
   445→        <div className="size-7 rounded-full bg-violet-100 dark:bg-violet-900/40 flex items-center justify-center">
   446→          <Sparkles className="size-4 text-violet-600 dark:text-violet-400" />
   447→        </div>
   448→        <span className="text-sm font-semibold">AI Resume Assistant</span>
   449→        <Badge variant="secondary" className="text-[10px] ml-auto">AI</Badge>
   450→      </div>
   451→
   452→      {/* Scrollable messages - min-h-0 is critical for flex overflow to work */}
   453→      <div
   454→        className="flex-1 min-h-0 overflow-y-auto px-4 py-3 scroll-smooth"
   455→        ref={scrollAreaRef}
   456→        style={{ scrollbarGutter: "stable" }}
   457→      >
   458→        <div className="space-y-3 pr-1">
   459→          {messages.map((msg, idx) => (
   460→            <div
   461→              key={idx}
   462→              className={`flex gap-2 ${msg.role === "user" ? "justify-end" : "justify-start"}`}
   463→            >
   464→              {msg.role === "assistant" && (
   465→                <div className="size-7 rounded-full bg-violet-100 dark:bg-violet-900/40 flex items-center justify-center shrink-0 mt-0.5">
   466→                  <Bot className="size-4 text-violet-600 dark:text-violet-400" />
   467→                </div>
   468→              )}
   469→              <div
   470→                className={`max-w-[85%] rounded-xl px-3.5 py-2.5 text-sm leading-relaxed ${
   471→                  msg.role === "user"
   472→                    ? "bg-primary text-primary-foreground"
   473→                    : "bg-muted"
   474→                }`}
   475→              >
   476→                <p className="whitespace-pre-wrap break-words">{msg.content}</p>
   477→              </div>
   478→            </div>
   479→          ))}
   480→          {isSending && (
   481→            <div className="flex gap-2 justify-start">
   482→              <div className="size-7 rounded-full bg-violet-100 dark:bg-violet-900/40 flex items-center justify-center shrink-0">
   483→                <Bot className="size-4 text-violet-600 dark:text-violet-400" />
   484→              </div>
   485→              <div className="bg-muted rounded-xl px-3.5 py-2.5">
   486→                <div className="flex items-center gap-2">
   487→                  <Loader2 className="size-4 animate-spin text-muted-foreground" />
   488→                  <span className="text-xs text-muted-foreground">Thinking...</span>
   489→                </div>
   490→              </div>
   491→            </div>
   492→          )}
   493→          {/* Invisible element to scroll to */}
   494→          <div ref={messagesEndRef} />
   495→        </div>
   496→      </div>
   497→
   498→      {/* Quick actions */}
   499→      {messages.length <= 1 && !isSending && (
   500→        <div className="px-4 pb-2 shrink-0 border-t pt-2">
   501→          <p className="text-[11px] text-muted-foreground mb-1.5 font-medium uppercase tracking-wider">Quick Actions</p>
   502→          <div className="flex flex-wrap gap-1.5">
   503→            {quickActions.map((action) => (
   504→              <Button
   505→                key={action.label}
   506→                variant="outline"
   507→                size="sm"
   508→                className="text-xs h-7 gap-1 text-violet-600 border-violet-200 hover:bg-violet-50 dark:text-violet-400 dark:border-violet-800 dark:hover:bg-violet-950"
   509→                onClick={() => handleSend(action.message)}
   510→              >
   511→                <Sparkles className="size-3" />
   512→                {action.label}
   513→              </Button>
   514→            ))}
   515→          </div>
   516→        </div>
   517→      )}
   518→
   519→      {/* Input area */}
   520→      <div className="flex gap-2 shrink-0 p-4 pt-2 border-t bg-card">
   521→        <Input
   522→          value={input}
   523→          onChange={(e) => setInput(e.target.value)}
   524→          onKeyDown={(e) => {
   525→            if (e.key === "Enter" && !e.shiftKey) {
   526→              e.preventDefault();
   527→              handleSend();
   528→            }
   529→          }}
   530→          placeholder="Ask AI to improve your resume..."
   531→          className="flex-1"
   532→          disabled={isSending}
   533→        />
   534→        <Button size="icon" onClick={() => handleSend()} disabled={isSending || !input.trim()}>
   535→          <Send className="size-4" />
   536→        </Button>
   537→      </div>
   538→    </div>
   539→  );
   540→}
