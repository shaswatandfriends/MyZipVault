    91→// ─── Status Helpers ───────────────────────────────────────────────────────
    92→
    93→type DisplayStatus = "pending" | "in_progress" | "completed";
    94→
    95→function getDisplayStatus(item: ChecklistItem): DisplayStatus {
    96→  if (item.status === "completed") return "completed";
    97→  if (item.status === "opened" || item.responseStatus === "active")
    98→    return "in_progress";
    99→  return "pending";
   100→}
   101→
   102→function getStatusBadge(status: DisplayStatus) {
   103→  switch (status) {
   104→    case "pending":
   105→      return (
   106→        <Badge className="bg-amber-50 border-amber-200 text-amber-700 hover:bg-amber-100">
   107→          <Clock className="size-3 mr-1" />
   108→          Pending
   109→        </Badge>
   110→      );
   111→    case "in_progress":
   112→      return (
   113→        <Badge className="bg-blue-50 border-blue-200 text-blue-700 hover:bg-blue-100">
   114→          <Loader2 className="size-3 mr-1" />
   115→          In Progress
   116→        </Badge>
   117→      );
   118→    case "completed":
   119→      return (
   120→        <Badge className="bg-green-50 border-green-200 text-green-700 hover:bg-green-100">
   121→          <CheckCircle2 className="size-3 mr-1" />
   122→          Completed
   123→        </Badge>
   124→      );
   125→  }
   126→}
   127→
   128→function getStatusColor(status: DisplayStatus) {
   129→  switch (status) {
   130→    case "pending":
   131→      return "text-amber-600";
   132→    case "in_progress":
   133→      return "text-blue-600";
   134→    case "completed":
   135→      return "text-green-600";
   136→  }
   137→}

   182→function RatingButton({
   183→  value,
   184→  label,
   185→  isSelected,
   186→  onClick,
   187→  disabled,
   188→}: {
   189→  value: string;
   190→  label: string;
   191→  isSelected: boolean;
   192→  onClick: () => void;
   193→  disabled: boolean;
   194→}) {
   195→  const style = ratingBtnStyles[value];
   196→  return (
   197→    <button
   198→      type="button"
   199→      disabled={disabled}
   200→      onClick={onClick}
   201→      className={`inline-flex items-center justify-center rounded-lg border-2 px-3 py-1.5 text-sm font-semibold transition-all duration-150 min-w-[44px] ${
   202→        isSelected ? style.selected : style.unselected
   203→      } ${disabled ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   204→      title={label}
   205→    >
   206→      {value}
   207→    </button>
   208→  );
   209→}

   211→// ─── Main Page Component ──────────────────────────────────────────────────
   212→
   213→export default function CandidateChecklistsPage() {
   214→  const [checklists, setChecklists] = useState<ChecklistItem[]>([]);
   215→  const [isLoading, setIsLoading] = useState(true);
   216→  const [error, setError] = useState("");
   217→
   218→  // Dialog states
   219→  const [formOpen, setFormOpen] = useState(false);
   220→  const [viewOpen, setViewOpen] = useState(false);
   221→  const [selectedChecklist, setSelectedChecklist] =
   222→    useState<ChecklistItem | null>(null);
   223→
   224→  // Form state
   225→  const [ratings, setRatings] = useState<RatingState>({});
   226→  const [signatureName, setSignatureName] = useState("");
   227→  const [isSaving, setIsSaving] = useState(false);
   228→  const [isSubmitting, setIsSubmitting] = useState(false);
   229→  const autoSaveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   230→
   231→  // ─── Fetch Checklists ─────────────────────────────────────────────────
   232→
   233→  const fetchChecklists = useCallback(async () => {
   234→    try {
   235→      const res = await fetch("/api/candidate/checklists");
   236→      if (!res.ok) throw new Error("Failed to fetch checklists");
   237→      const data = await res.json();
   238→      setChecklists(data.checklists || []);
   239→    } catch {
   240→      setError("Failed to load checklists. Please refresh the page.");
   241→    } finally {
   242→      setIsLoading(false);
   243→    }
   244→  }, []);
   245→
   246→  useEffect(() => {
   247→    fetchChecklists();
   248→  }, [fetchChecklists]);
   249→
   250→  // ─── Stats ────────────────────────────────────────────────────────────
   251→
   252→  const stats = useMemo(() => {
   253→    const pending = checklists.filter(
   254→      (c) => getDisplayStatus(c) === "pending"
   255→    ).length;
   256→    const inProgress = checklists.filter(
   257→      (c) => getDisplayStatus(c) === "in_progress"
   258→    ).length;
   259→    const completed = checklists.filter(
   260→      (c) => getDisplayStatus(c) === "completed"
   261→    ).length;
   262→    return { pending, inProgress, completed };
   263→  }, [checklists]);
   264→
   265→  // ─── Auto-Save ────────────────────────────────────────────────────────
   266→
   267→  const autoSave = useCallback(
   268→    async (
   269→      requestId: number,
   270→      updatedRatings: RatingState,
   271→      skills: Skill[]
   272→    ) => {
   273→      try {
   274→        const ratingPayload = skills.map((skill) => ({
   275→          skillId: skill.id,
   276→          ratingValue:
   277→            skill.questionType === "text"
   278→              ? updatedRatings[skill.id]?.textValue || null
   279→              : updatedRatings[skill.id]?.ratingValue || null,
   280→          isNa: updatedRatings[skill.id]?.isNa || false,
   281→        }));
   282→
   283→        const res = await fetch("/api/candidate/checklists/submit", {
   284→          method: "POST",
   285→          headers: { "Content-Type": "application/json" },
   286→          body: JSON.stringify({
   287→            requestId,
   288→            ratings: ratingPayload,
   289→          }),
   290→        });
   291→
   292→        if (res.ok) {
   293→          const result = await res.json();
   294→          // Update the checklist's completion percentage locally
   295→          setChecklists((prev) =>
   296→            prev.map((c) =>
   297→              c.id === requestId
   298→                ? { ...c, completionPct: result.completionPct }
   299→                : c
   300→            )
   301→          );
   302→        }
   303→      } catch {
   304→        // Silent fail for auto-save
   305→      }
   306→    },
   307→    []
   308→  );
   309→
   310→  const debouncedAutoSave = useCallback(
   311→    (requestId: number, updatedRatings: RatingState, skills: Skill[]) => {
   312→      if (autoSaveTimerRef.current) {
   313→        clearTimeout(autoSaveTimerRef.current);
   314→      }
   315→      autoSaveTimerRef.current = setTimeout(() => {
   316→        autoSave(requestId, updatedRatings, skills);
   317→      }, 800);
   318→    },
   319→    [autoSave]
   320→  );
   321→
   322→  // ─── Rating Change Handler ────────────────────────────────────────────
   323→
   324→  const handleRatingChange = useCallback(
   325→    (
   326→      skillId: number,
   327→      value: string | null,
   328→      isNa: boolean,
   329→      textValue?: string
   330→    ) => {
   331→      if (!selectedChecklist) return;
   332→
   333→      setRatings((prev) => {
   334→        const updated = {
   335→          ...prev,
   336→          [skillId]: { ratingValue: value, isNa, textValue },
   337→        };
   338→
   339→        // Trigger auto-save
   340→        debouncedAutoSave(
   341→          selectedChecklist.id,
   342→          updated,
   343→          selectedChecklist.template.skills
   344→        );
   345→
   346→        return updated;
   347→      });
   348→    },
   349→    [selectedChecklist, debouncedAutoSave]
   350→  );
   351→
   352→  // ─── Open Checklist Dialog ────────────────────────────────────────────
   353→
   354→  const handleOpenChecklist = useCallback((checklist: ChecklistItem) => {
   355→    setSelectedChecklist(checklist);
   356→
   357→    // Initialize ratings from existing ratings
   358→    const initialRatings: RatingState = {};
   359→    checklist.existingRatings.forEach((r) => {
   360→      initialRatings[r.skillId] = {
   361→        ratingValue: r.ratingValue,
   362→        isNa: r.isNa,
   363→        textValue: r.ratingValue || "",
   364→      };
   365→    });
   366→    setRatings(initialRatings);
   367→    setSignatureName("");
   368→    setFormOpen(true);
   369→  }, []);
   370→
   371→  // ─── View Submission Dialog ───────────────────────────────────────────
   372→
   373→  const handleViewSubmission = useCallback((checklist: ChecklistItem) => {
   374→    setSelectedChecklist(checklist);
   375→
   376→    const initialRatings: RatingState = {};
   377→    checklist.existingRatings.forEach((r) => {
   378→      initialRatings[r.skillId] = {
   379→        ratingValue: r.ratingValue,
   380→        isNa: r.isNa,
   381→        textValue: r.ratingValue || "",
   382→      };
   383→    });
   384→    setRatings(initialRatings);
   385→    setViewOpen(true);
   386→  }, []);
   387→
   388→  // ─── Calculate Completion ─────────────────────────────────────────────
   389→
   390→  const calculateCompletion = useCallback(
   391→    (checklist: ChecklistItem, currentRatings?: RatingState) => {
   392→      const skills = checklist.template.skills;
   393→      if (skills.length === 0) return 0;
   394→
   395→      const ratedCount = skills.filter((skill) => {
   396→        const rating =
   397→          currentRatings?.[skill.id] ??
   398→          checklist.existingRatings.find((r) => r.skillId === skill.id);
   399→        if (!rating) return false;
   400→        if (rating.isNa) return true;
   401→        if (skill.questionType === "text") return !!rating.textValue?.trim();
   402→        return !!rating.ratingValue;
   403→      }).length;
   404→
   405→      return Math.round((ratedCount / skills.length) * 100);
   406→    },
   407→    []
   408→  );
   409→
   410→  const isAllSkillsRated = useMemo(() => {
   411→    if (!selectedChecklist) return false;
   412→    return calculateCompletion(selectedChecklist, ratings) === 100;
   413→  }, [selectedChecklist, ratings, calculateCompletion]);
   414→
   415→  // ─── Final Submit ─────────────────────────────────────────────────────
   416→
   417→  const handleSubmit = useCallback(async () => {
   418→    if (!selectedChecklist || !signatureName.trim()) return;
   419→
   420→    setIsSubmitting(true);
   421→    try {
   422→      const skills = selectedChecklist.template.skills;
   423→      const ratingPayload = skills.map((skill) => ({
   424→        skillId: skill.id,
   425→        ratingValue:
   426→          skill.questionType === "text"
   427→            ? ratings[skill.id]?.textValue || null
   428→            : ratings[skill.id]?.ratingValue || null,
   429→        isNa: ratings[skill.id]?.isNa || false,
   430→      }));
   431→
   432→      const res = await fetch("/api/candidate/checklists/submit", {
   433→        method: "POST",
   434→        headers: { "Content-Type": "application/json" },
   435→        body: JSON.stringify({
   436→          requestId: selectedChecklist.id,
   437→          ratings: ratingPayload,
   438→          digitalSignature: signatureName.trim(),
   439→          candidateNameSigned: signatureName.trim(),
   440→        }),
   441→      });
   442→
   443→      if (!res.ok) {
   444→        const data = await res.json();
   445→        throw new Error(data.error || "Failed to submit checklist");
   446→      }
   447→
   448→      toast.success("Checklist submitted successfully!", {
   449→        description: "Your skills checklist has been saved and submitted.",
   450→      });
   451→
   452→      setFormOpen(false);
   453→      setSelectedChecklist(null);
   454→      fetchChecklists();
   455→    } catch (err) {
   456→      toast.error("Failed to submit checklist", {
   457→        description:
   458→          err instanceof Error ? err.message : "Please try again.",
   459→      });
   460→    } finally {
   461→      setIsSubmitting(false);
   462→    }
   463→  }, [selectedChecklist, signatureName, ratings, fetchChecklists]);
   464→
   465→  // ─── Group Skills by Category ─────────────────────────────────────────
   466→
   467→  const groupSkillsByCategory = (skills: Skill[]) => {
   468→    const groups: Record<string, Skill[]> = {};
   469→    skills.forEach((skill) => {
   470→      if (!groups[skill.category]) {
   471→        groups[skill.category] = [];
   472→      }
   473→      groups[skill.category].push(skill);
   474→    });
   475→    return groups;
   476→  };
   477→
   478→  // ─── Rating Legend Component ──────────────────────────────────────────
   479→
   480→  const RatingLegend = () => (
   481→    <div className="rounded-lg border border-gray-200 bg-gray-50 p-4">
   482→      <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider mb-3">
   483→        Rating Scale
   484→      </p>
   485→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   486→        {(["1", "2", "3", "4"] as const).map((val) => {
   487→          const style = ratingBtnStyles[val];
   488→          return (
   489→            <div key={val} className="flex items-center gap-2">
   490→              <span
   491→                className={`inline-flex items-center justify-center rounded-lg border-2 px-2 py-1 text-xs font-bold ${style.selected}`}
   492→              >
   493→                {val}
   494→              </span>
   495→              <span className="text-xs text-gray-600">
   496→                {ratingLabels[val]}
   497→              </span>
   498→            </div>
   499→          );
   500→        })}
   501→      </div>
   502→    </div>
   503→  );
   504→
   505→  // ─── Skill Row Component ──────────────────────────────────────────────
   506→
   507→  const SkillRow = ({
   508→    skill,
   509→    disabled,
   510→  }: {
   511→    skill: Skill;
   512→    disabled: boolean;
   513→  }) => {
   514→    const currentRating = ratings[skill.id];
   515→
   516→    if (skill.questionType === "yes_no") {
   517→      const currentVal = currentRating?.ratingValue;
   518→      const isNa = currentRating?.isNa ?? false;
   519→
   520→      return (
   521→        <div className="flex items-center justify-between gap-3 py-3 border-b border-gray-100 last:border-b-0">
   522→          <span
   523→            className={`text-sm font-medium flex-1 ${
   524→              isNa ? "text-gray-400 line-through" : "text-gray-700"
   525→            }`}
   526→          >
   527→            {skill.skillName}
   528→          </span>
   529→          <div className="flex items-center gap-2">
   530→            <button
   531→              type="button"
   532→              disabled={disabled}
   533→              onClick={() =>
   534→                handleRatingChange(skill.id, "yes", false)
   535→              }
   536→              className={`inline-flex items-center justify-center rounded-lg border-2 px-4 py-1.5 text-sm font-semibold transition-all duration-150 min-w-[52px] ${
   537→                currentVal === "yes" && !isNa
   538→                  ? "bg-[#166534] border-[#166534] text-white shadow-sm"
   539→                  : "border-gray-200 text-gray-400 hover:border-[#166534] hover:text-[#166534]"
   540→              } ${disabled ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   541→            >
   542→              Yes
   543→            </button>
   544→            <button
   545→              type="button"
   546→              disabled={disabled}
   547→              onClick={() =>
   548→                handleRatingChange(skill.id, "no", false)
   549→              }
   550→              className={`inline-flex items-center justify-center rounded-lg border-2 px-4 py-1.5 text-sm font-semibold transition-all duration-150 min-w-[52px] ${
   551→                currentVal === "no" && !isNa
   552→                  ? "bg-[#DC2626] border-[#DC2626] text-white shadow-sm"
   553→                  : "border-gray-200 text-gray-400 hover:border-[#DC2626] hover:text-[#DC2626]"
   554→              } ${disabled ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   555→            >
   556→              No
   557→            </button>
   558→            {skill.hasNaOption && (
   559→              <button
   560→                type="button"
   561→                disabled={disabled}
   562→                onClick={() =>
   563→                  handleRatingChange(
   564→                    skill.id,
   565→                    isNa ? null : currentVal,
   566→                    !isNa
   567→                  )
   568→                }
   569→                className={`inline-flex items-center justify-center rounded-lg border-2 px-3 py-1.5 text-xs font-semibold transition-all duration-150 min-w-[44px] ${
   570→                  isNa
   571→                    ? "bg-gray-600 border-gray-600 text-white shadow-sm"
   572→                    : "border-gray-200 text-gray-400 hover:border-gray-500 hover:text-gray-600"
   573→                } ${disabled ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   574→              >
   575→                N/A
   576→              </button>
   577→            )}
   578→          </div>
   579→        </div>
   580→      );
   581→    }
   582→
   583→    if (skill.questionType === "text") {
   584→      const textVal = currentRating?.textValue || "";
   585→      const isNa = currentRating?.isNa ?? false;
   586→
   587→      return (
   588→        <div className="py-3 border-b border-gray-100 last:border-b-0">
   589→          <Label
   590→            className={`text-sm font-medium mb-2 block ${
   591→              isNa ? "text-gray-400 line-through" : "text-gray-700"
   592→            }`}
   593→          >
   594→            {skill.skillName}
   595→          </Label>
   596→          <div className="flex items-start gap-2">
   597→            <Textarea
   598→              value={isNa ? "" : textVal}
   599→              onChange={(e) =>
   600→                handleRatingChange(skill.id, null, false, e.target.value)
   601→              }
   602→              disabled={disabled || isNa}
   603→              placeholder="Enter your response..."
   604→              className="min-h-[60px] resize-none"
   605→            />
   606→            {skill.hasNaOption && (
   607→              <button
   608→                type="button"
   609→                disabled={disabled}
   610→                onClick={() =>
   611→                  handleRatingChange(skill.id, null, !isNa, textVal)
   612→                }
   613→                className={`inline-flex items-center justify-center rounded-lg border-2 px-3 py-2 text-xs font-semibold transition-all duration-150 min-w-[44px] shrink-0 ${
   614→                  isNa
   615→                    ? "bg-gray-600 border-gray-600 text-white shadow-sm"
   616→                    : "border-gray-200 text-gray-400 hover:border-gray-500 hover:text-gray-600"
   617→                } ${disabled ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   618→              >
   619→                N/A
   620→              </button>
   621→            )}
   622→          </div>
