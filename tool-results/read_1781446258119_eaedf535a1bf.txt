    93→
    94→function FadeInOnScroll({ children, className }: { children: React.ReactNode; className?: string }) {
    95→  const ref = useRef<HTMLDivElement>(null);
    96→  const [isVisible, setIsVisible] = useState(false);
    97→
    98→  useEffect(() => {
    99→    const observer = new IntersectionObserver(
   100→      ([entry]) => {
   101→        if (entry.isIntersecting) {
   102→          setIsVisible(true);
   103→          observer.unobserve(entry.target);
   104→        }
   105→      },
   106→      { threshold: 0.1 }
   107→    );
   108→    if (ref.current) observer.observe(ref.current);
   109→    return () => observer.disconnect();
   110→  }, []);
   111→
   112→  return (
   113→    <motion.div
   114→      ref={ref}
   115→      initial={{ opacity: 0, y: 24 }}
   116→      animate={isVisible ? { opacity: 1, y: 0 } : { opacity: 0, y: 24 }}
   117→      transition={{ duration: 0.6, ease: "easeOut" }}
   118→      className={className}
   119→    >
   120→      {children}
   121→    </motion.div>
   122→  );
   123→}
