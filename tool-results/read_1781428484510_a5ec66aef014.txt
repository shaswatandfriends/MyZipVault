    99→// ─── Helper: extract {{variables}} from text ────────────────────────
   100→function extractVariables(text: string): string[] {
   101→  const matches = text.match(/\{\{(\w+)\}\}/g);
   102→  if (!matches) return [];
   103→  return [...new Set(matches.map((m) => m.replace(/\{\{|\}\}/g, "")))];
   104→}
   105→
   106→// ─── Helper: replace variables with sample data ─────────────────────
   107→function renderWithSampleData(text: string): string {
   108→  let result = text;
   109→  const variables = extractVariables(text);
   110→  for (const v of variables) {
   111→    if (SAMPLE_DATA[v]) {
   112→      result = result.replaceAll(`{{${v}}}`, SAMPLE_DATA[v]);
   113→    }
   114→  }
   115→  return result;
   116→}
